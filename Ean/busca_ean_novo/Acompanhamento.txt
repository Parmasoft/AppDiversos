class VincularCodigoPage extends StatefulWidget {
  final String codigoBarras;
  final String baseUrl;

  const VincularCodigoPage({
    super.key,
    required this.codigoBarras,
    required this.baseUrl,
  });

  @override
  State<VincularCodigoPage> createState() => _VincularCodigoPageState();

class _VincularCodigoPageState extends State<VincularCodigoPage> {
  final TextEditingController _buscaCtrl = TextEditingController();

  bool _carregando = false;
  bool _gravando = false;
  bool _buscandoFoto = false;
  String _mensagem = "";
  String _textoFotoLido = "";
  List<dynamic> _produtos = [];
  Map<String, dynamic>? _selecionado;
  Timer? _debounceBusca;

  @override
  void dispose() {
    _debounceBusca?.cancel();
    _buscaCtrl.dispose();
    super.dispose();
  }

  String _limparTextoFoto(String texto) {
    var t = texto.toUpperCase();

    const acentos = {
      'Á': 'A',
      'À': 'A',
      'Â': 'A',
      'Ã': 'A',
      'É': 'E',
      'Ê': 'E',
      'Í': 'I',
      'Ó': 'O',
      'Ô': 'O',
      'Õ': 'O',
      'Ú': 'U',
      'Ü': 'U',
      'Ç': 'C',
    };

    acentos.forEach((de, para) {
      t = t.replaceAll(de, para);
    });

    for (final ch in ['.', ',', '-', '_', '/', '\\', '(', ')', '%', ':', ';']) {
      t = t.replaceAll(ch, ' ');
    }

    final ignorar = <String>{
      'DE',
      'DA',
      'DO',
      'DAS',
      'DOS',
      'COM',
      'PARA',
      'POR',
      'EM',
      'PRODUTO',
      'USO',
      'ANTES',
      'LEIA',
      'INSTRUCOES',
      'ROTULO',
      'ATENCAO',
      'PERIGO',
      'MANTENHA',
      'FORA',
      'ALCANCE',
      'CRIANCA',
      'CRIANÇAS',
      'INTEGRAL',
      'ORIGINAL',
      'REGISTRO',
      'VALIDADE',
      'LOTE',
      'FABRICACAO',
      'FABRICAÇÃO',
    };

    final palavras = t
        .split(RegExp(r'\s+'))
        .map((p) => p.trim())
        .where((p) => p.length >= 2)
        .where((p) => !ignorar.contains(p))
        .toList();

    final unicas = <String>[];

    for (final p in palavras) {
      if (!unicas.contains(p)) {
        unicas.add(p);
      }
    }

    return unicas.take(10).join(' ');
  }

  void _aoDigitarBusca(String valor) {
    _debounceBusca?.cancel();

    final termo = valor.trim();

    if (termo.length < 3) {
      setState(() {
        _mensagem = "Digite pelo menos 3 caracteres para buscar.";
        _produtos = [];
        _selecionado = null;
      });
      return;
    }

    _debounceBusca = Timer(const Duration(milliseconds: 600), () {
      _buscar();
    });
  }

  String _formatarPreco(dynamic valor) {
    final numero = double.tryParse(valor.toString()) ?? 0.0;
    final formatter = NumberFormat.currency(
      locale: 'pt_BR',
      symbol: 'R\$',
      decimalDigits: 2,
    );
    return formatter.format(numero);
  }

  Future<void> _buscarPorFoto() async {
    FocusScope.of(context).unfocus();

    setState(() {
      _buscandoFoto = true;
      _mensagem = "";
      _textoFotoLido = "";
      _produtos = [];
      _selecionado = null;
    });

    try {
      final picker = ImagePicker();

      final foto = await picker.pickImage(
        source: ImageSource.camera,
        imageQuality: 85,
        maxWidth: 1400,
      );

      if (foto == null) {
        if (!mounted) return;
        setState(() {
          _buscandoFoto = false;
        });
        return;
      }

      final inputImage = InputImage.fromFilePath(foto.path);
      final textRecognizer = TextRecognizer(
        script: TextRecognitionScript.latin,
      );

      final recognizedText = await textRecognizer.processImage(inputImage);
      await textRecognizer.close();

      final textoLido = recognizedText.text.trim();
      final termoBusca = _limparTextoFoto(textoLido);

      if (!mounted) return;

      if (termoBusca.length < 3) {
        setState(() {
          _buscandoFoto = false;
          _textoFotoLido = textoLido;
          _mensagem = "Não consegui identificar texto suficiente na foto.";
        });
        return;
      }

      setState(() {
        _buscandoFoto = false;
        _textoFotoLido = textoLido;
        _buscaCtrl.text = termoBusca;
      });

      await _buscar();
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _buscandoFoto = false;
        _mensagem = "Erro ao ler a foto do produto.";
      });
    }
  }

  Future<void> _buscar() async {
    final termo = _buscaCtrl.text.trim();

    if (termo.length < 3) {
      setState(() {
        _mensagem = "Digite pelo menos 3 caracteres para buscar.";
        _produtos = [];
        _selecionado = null;
      });
      return;
    }

    setState(() {
      _carregando = true;
      _mensagem = "";
      _produtos = [];
      _selecionado = null;
    });

    try {
      final url = Uri.parse(
        "${widget.baseUrl}/produto/buscar-descricao?empresa=1&termo=${Uri.encodeComponent(termo)}",
      );

      final resp = await http.get(url).timeout(const Duration(seconds: 8));

      if (!mounted) return;

      if (resp.statusCode != 200) {
        setState(() {
          _mensagem = "Erro ao buscar produtos.";
        });
        return;
      }

      final json = jsonDecode(resp.body);
      final produtos = json["produtos"] ?? [];

      setState(() {
        _produtos = produtos;
        _mensagem = produtos.isEmpty
            ? "Nenhum produto encontrado com essa busca."
            : "";
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _mensagem = "Erro de conexão com o servidor.";
      });
    } finally {
      if (mounted) {
        setState(() => _carregando = false);
      }
    }
  }

  Future<void> _confirmarVinculo() async {
    if (_selecionado == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text("Selecione um produto primeiro."),
          backgroundColor: Colors.orange,
        ),
      );
      return;
    }

    final produtoId = _selecionado!["produto_id"];
    final descricao = _selecionado!["descricao"] ?? "";

    final confirmar = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text("Confirmar vínculo"),
        content: Text(
          "Deseja vincular o código:\n\n"
          "${widget.codigoBarras}\n\n"
          "ao produto:\n\n"
          "$produtoId - $descricao ?",
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text("Não"),
          ),
          ElevatedButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text("Sim, vincular"),
          ),
        ],
      ),
    );

    if (confirmar != true) return;

    setState(() => _gravando = true);

    try {
      final resp = await http
          .post(
            Uri.parse("${widget.baseUrl}/produto/vincular-codigo"),
            headers: {"Content-Type": "application/json"},
            body: jsonEncode({
              "empresa": 1,
              "produto_id": produtoId,
              "codigo_barras": widget.codigoBarras,
            }),
          )
          .timeout(const Duration(seconds: 8));

      final json = jsonDecode(resp.body);

      if (!mounted) return;

      if (resp.statusCode == 200 && json["status"] == "ok") {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(json["mensagem"] ?? "Código vinculado com sucesso."),
            backgroundColor: Colors.green,
          ),
        );

        Navigator.pop(context, true);
      } else {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(json["mensagem"] ?? "Erro ao vincular código."),
            backgroundColor: Colors.red,
          ),
        );
      }
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text("Erro de conexão ao vincular código."),
          backgroundColor: Colors.red,
        ),
      );
    } finally {
      if (mounted) {
        setState(() => _gravando = false);
      }
    }
  }

class _LogoGiraState extends State<LogoGira>
    with SingleTickerProviderStateMixin {
  late AnimationController controller;

  @override
  void initState() {
    super.initState();
    controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 10),
    )..repeat();
  }

  @override
  void dispose() {
    controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: controller,
      builder: (_, child) {
        final angle = controller.value * 6.28318;
        return Transform(
          alignment: Alignment.center,
          transform: Matrix4.identity()
            ..setEntry(3, 2, 0.001)
            ..rotateY(-angle),
          child: child,
        );
      },
      child: ClipRRect(
        borderRadius: BorderRadius.circular(35),
        child: Image.asset(
          'assets/images/logo.jpeg',
          width: 180,
          fit: BoxFit.contain,
        ),

class _TelaUnicaState extends State<TelaUnica> {
  // Default (caso não exista config salva)
  String baseUrl = "http://192.168.0.250:5001";
  //String baseUrl = "http://192.168.1.238:5001";

  @override
  void initState() {
    super.initState();
    _carregarBaseUrl();
  }

  Future<void> _carregarBaseUrl() async {
    final prefs = await SharedPreferences.getInstance();
    final saved = prefs.getString("api_base_url");
    if (saved != null && saved.trim().isNotEmpty) {
      setState(() => baseUrl = saved.trim());
    }
  }

  // --------------------------------------------------------------------------
  // ABRIR CONFIG
  // --------------------------------------------------------------------------
  Future<void> abrirConfig() async {
    final novo = await Navigator.push<String>(
      context,
      MaterialPageRoute(builder: (_) => ConfigPage(baseUrlAtual: baseUrl)),
    );

    if (novo != null && novo.trim().isNotEmpty) {
      setState(() => baseUrl = novo.trim());
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text("Configuração salva: $baseUrl"),
          backgroundColor: Colors.green,
        ),
      );
    }
  }

  // --------------------------------------------------------------------------
  // ABRIR SCANNER
  // --------------------------------------------------------------------------
  Future<void> abrirScanner() async {
    final String? codigo = await Navigator.push<String>(
      context,
      MaterialPageRoute(builder: (_) => const ScannerPage()),
    );

    if (codigo == null || codigo.trim().isEmpty) return;

    final resultado = await buscar(codigo.trim());

    if (!mounted) return;

    switch (resultado["status"]) {
      case "ok":
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (_) => ProdutoEncontradoPage(
              dados: resultado["dados"],
              baseUrl: baseUrl, // 👈 A CONFIGURAÇÃO ATUAL
              onBuscar: () {
                Navigator.pop(context);
                abrirScanner();
              },
            ),
          ),
        );
        break;

      case "nao_encontrado":
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Produto não encontrado"),
            backgroundColor: Colors.orange,
          ),
        );
        break;

      case "erro_conexao":
      default:
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Erro de conexão com o servidor"),
            backgroundColor: Colors.red,
          ),
        );
        break;
    }
  }

  // --------------------------------------------------------------------------
  // BUSCAR PRODUTO NA API (COM STATUS)
  // --------------------------------------------------------------------------
  Future<Map<String, dynamic>> buscar(String cod) async {
    try {
      final url = Uri.parse("$baseUrl/produto?cod=$cod");
      final resp = await http.get(url).timeout(const Duration(seconds: 5));

      if (resp.statusCode != 200) {
        return {"status": "erro_conexao"};
      }

      final dados = jsonDecode(resp.body);

      if (dados["encontrado"] == true ||
          dados["encontrado"] == 1 ||
          dados["encontrado"] == "true") {
        return {
          "status": "ok",
          "dados": {
            "produto_id": dados["produto_id"],
            "codigo": dados["codigo"] ?? cod,
            "descricao": dados["descricao"] ?? "",
            "unidade": dados["unidade"] ?? "",
            "preco": dados["preco"]?.toString() ?? "0.00",
          },
        };
      } else {
        return {"status": "nao_encontrado"};
      }
    } catch (_) {
      return {"status": "erro_conexao"};
    }
  }

  State<LogoGira> createState() => _LogoGiraState();

class _LogoGiraState extends State<LogoGira>
    with SingleTickerProviderStateMixin {
  late AnimationController controller;

  @override
  void initState() {
    super.initState();
    controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 10),
    )..repeat();
  }

  @override
  void dispose() {
    controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: controller,
      builder: (_, child) {
        final angle = controller.value * 6.28318;
        return Transform(
          alignment: Alignment.center,
          transform: Matrix4.identity()
            ..setEntry(3, 2, 0.001)
            ..rotateY(-angle),
          child: child,
        );
      },
      child: ClipRRect(
        borderRadius: BorderRadius.circular(35),
        child: Image.asset(
          'assets/images/logo.jpeg',
          width: 180,
          fit: BoxFit.contain,
        ),
      ),
    );

class TelaUnica extends StatefulWidget {
  const TelaUnica({super.key});

  @override
  State<TelaUnica> createState() => _TelaUnicaState();
}

class _TelaUnicaState extends State<TelaUnica> {
  // Default (caso não exista config salva)
  String baseUrl = "http://192.168.0.250:5001";
  //String baseUrl = "http://192.168.1.238:5001";

  @override
  void initState() {
    super.initState();
    _carregarBaseUrl();
  }

  Future<void> _carregarBaseUrl() async {
    final prefs = await SharedPreferences.getInstance();
    final saved = prefs.getString("api_base_url");
    if (saved != null && saved.trim().isNotEmpty) {
      setState(() => baseUrl = saved.trim());
    }
  }

  // --------------------------------------------------------------------------
  // ABRIR CONFIG
  // --------------------------------------------------------------------------
  Future<void> abrirConfig() async {
    final novo = await Navigator.push<String>(
      context,
      MaterialPageRoute(builder: (_) => ConfigPage(baseUrlAtual: baseUrl)),
    );

    if (novo != null && novo.trim().isNotEmpty) {
      setState(() => baseUrl = novo.trim());
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text("Configuração salva: $baseUrl"),
          backgroundColor: Colors.green,
        ),
      );
    }
  }

  // --------------------------------------------------------------------------
  // ABRIR SCANNER
  // --------------------------------------------------------------------------
  Future<void> abrirScanner() async {
    final String? codigo = await Navigator.push<String>(
      context,
      MaterialPageRoute(builder: (_) => const ScannerPage()),
    );

    if (codigo == null || codigo.trim().isEmpty) return;

    final resultado = await buscar(codigo.trim());

    if (!mounted) return;

    switch (resultado["status"]) {
      case "ok":
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (_) => ProdutoEncontradoPage(
              dados: resultado["dados"],
              baseUrl: baseUrl, // 👈 A CONFIGURAÇÃO ATUAL
              onBuscar: () {
                Navigator.pop(context);
                abrirScanner();
              },
            ),
          ),
        );
        break;

      case "nao_encontrado":
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Produto não encontrado"),
            backgroundColor: Colors.orange,
          ),
        );
        break;

      case "erro_conexao":
      default:
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Erro de conexão com o servidor"),
            backgroundColor: Colors.red,
          ),
        );
        break;
    }
  }

  // --------------------------------------------------------------------------
  // BUSCAR PRODUTO NA API (COM STATUS)
  // --------------------------------------------------------------------------
  Future<Map<String, dynamic>> buscar(String cod) async {
    try {
      final url = Uri.parse("$baseUrl/produto?cod=$cod");
      final resp = await http.get(url).timeout(const Duration(seconds: 5));

      if (resp.statusCode != 200) {
        return {"status": "erro_conexao"};
      }

      final dados = jsonDecode(resp.body);

      if (dados["encontrado"] == true ||
          dados["encontrado"] == 1 ||
          dados["encontrado"] == "true") {
        return {
          "status": "ok",
          "dados": {
            "produto_id": dados["produto_id"],
            "codigo": dados["codigo"] ?? cod,
            "descricao": dados["descricao"] ?? "",
            "unidade": dados["unidade"] ?? "",
            "preco": dados["preco"]?.toString() ?? "0.00",
          },
        };
      } else {
        return {"status": "nao_encontrado"};
      }
    } catch (_) {
      return {"status": "erro_conexao"};
    }
  }

  // --------------------------------------------------------------------------
  // UI
  // --------------------------------------------------------------------------
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.white,

      // RODAPÉ
      bottomNavigationBar: SizedBox(
        height: 70,
        child: Row(
          children: [
            botaoRodape(
              icon: Icons.camera_alt,
              texto: "Buscar",
              onTap: abrirScanner,
            ),
            botaoRodape(
              icon: Icons.settings,
              texto: "Configurar",
              onTap: abrirConfig,
            ),
          ],
        ),
      ),

      body: Column(
        children: [
          Container(
            height: MediaQuery.of(context).size.height * 0.35,
            width: double.infinity,
            decoration: const BoxDecoration(
              gradient: LinearGradient(
                colors: [Color(0xFFFFF9E3), Color(0xFFF7EFC7)],
                begin: Alignment.topCenter,
                end: Alignment.bottomCenter,
              ),
            ),
            child: const Center(child: LogoGira()),
          ),
          Expanded(
            child: Container(
              width: double.infinity,
              decoration: const BoxDecoration(
                gradient: LinearGradient(
                  colors: [Color(0xFFEDEDED), Color(0xFFD9D9D9)],
                  begin: Alignment.topCenter,
                  end: Alignment.bottomCenter,
                ),
              ),
              child: Center(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    const Text(
                      "Scanner ParmaSoft",

                        fontSize: 26,
