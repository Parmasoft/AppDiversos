def round_money(value):
    try:
        return round(float(value), 2)
    except Exception:
        return 0.00


def parse_float(value, default=0.0):
    try:
        return float(value)
    except Exception:
        return default


def parse_date(value):
    """
    Aceita:
      - 'YYYY-MM-DD' (padrão comum do Flutter)
      - 'DD/MM/YYYY' (padrão BR)
      - date (já convertido)
    """
    if value is None:
        return None
    if isinstance(value, date):
        return value

    s = str(value).strip()
    if not s:
        return None

    for fmt in ("%Y-%m-%d", "%d/%m/%Y"):
        try:
            return datetime.strptime(s, fmt).date()
        except Exception:
            pass

    # Se não conseguir converter, retorna como string mesmo (Firebird às vezes aceita),
    # mas é melhor mandar no formato 'YYYY-MM-DD' do app.
    return s


def parse_time(value):
    """
    Aceita:
      - 'HH:MM'
      - 'HH:MM:SS'
      - 'HH:MM AM/PM' ou 'HH:MM PM'
      - time (já convertido)
    """
    if value is None:
        return None
    if isinstance(value, time):
        return value

    s = str(value).strip()
    if not s:
        return None

    for fmt in ("%H:%M", "%H:%M:%S", "%I:%M %p"):
        try:
            return datetime.strptime(s, fmt).time()
        except Exception:
            pass

    return None

@app.route("/pedidos", methods=["POST"])
def salvar_pedido():
    data = request.json or {}

    pedido = data.get("pedido")
    itens = data.get("itens", [])

    if not pedido or not itens:
        return jsonify({"erro": "Pedido ou itens não informados"}), 400

    con = None
    try:
        con = get_connection()
        cur = con.cursor()

        # =============================
        # GERA PED_ID
        # =============================
        cur.execute("SELECT NEXT VALUE FOR GEN_TB_PEDIDO_PADARIA FROM RDB$DATABASE")
        pedido_id = cur.fetchone()[0]

        # =============================
        # DATA / HORA
        # =============================
        data_entrega = parse_date(pedido.get("data_entrega"))
        hora_entrega = parse_time(pedido.get("hora_entrega"))

        # =============================
        # INSERE CABEÇALHO
        # =============================
        cur.execute("""
            INSERT INTO TB_PEDIDO_PADARIA (
                PED_ID,
                PED_CLIENTE,
                PED_CONTATO,
                PED_PAGAMENTO,
                PED_DATA_ENT,
                PED_HORA_ENT,
                PED_OBS,
                PED_TOTAL,
                PED_STATUS,
                PED_DATA_INC
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)
        """, (
            pedido_id,
            pedido.get("cliente"),
            pedido.get("contato"),
            pedido.get("forma_pagamento"),
            data_entrega,
            hora_entrega,
            pedido.get("observacoes"),
            round_money(pedido.get("total")),
            "ENVIADO"
        ))

        # =============================
        # INSERE ITENS (PRODUTOS + BOLOS)
        #  - BOLO:
        #      ITEM_DESC1 = descrição do bolo
        #      ITEM_DESC2 = recheio 1
        #      ITEM_DESC3 = recheio 2 (opcional)  <-- AQUI
        #      ITEM_OBS   = observação do bolo
        # =============================
        for item in itens:
            cur.execute("SELECT NEXT VALUE FOR GEN_TB_PEDIDO_PADARIA_ITEM FROM RDB$DATABASE")
            item_id = cur.fetchone()[0]

            item_tipo = (item.get("item_tipo") or "").strip().upper()  # 'P' ou 'B'

            cur.execute("""
                INSERT INTO TB_PEDIDO_PADARIA_ITEM (
                    ITEM_ID,
                    PED_ID,
                    ITEM_DESC1,
                    ITEM_DESC2,
                    ITEM_DESC3,
                    ITEM_UNIDADE,
                    ITEM_QTD,
                    ITEM_PRECO,
                    ITEM_SUBTOTAL,
                    ITEM_TIPO,
                    REC_ID ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                item_id,
                pedido_id,
                item.get("descricao1"),
                item.get("descricao2"),
                item.get("descricao3"),  # <- recheio 2 (quando bolo)
                item.get("unidade"),
                parse_float(item.get("quantidade")),
                round_money(item.get("preco_unitario")),
                round_money(item.get("subtotal")),
                item_tipo,
                item.get("rec_id") if item_tipo == "B" else None,
))

        con.commit()

        return jsonify({
            "status": "ok",
            "pedido_id": pedido_id
        })

    except Exception as e:
        if con:
            con.rollback()
        print("Erro salvar pedido:", e)
        return jsonify({"erro": "Erro ao salvar pedido"}), 500
    finally:
        if con:
            con.close()

@app.route("/upload/pedido-foto", methods=["POST"])
def upload_pedido_foto():
    ped_id = request.form.get("ped_id")
    file = request.files.get("foto")

    if not ped_id:
        return jsonify({"ok": False, "msg": "ped_id não informado"}), 400
    if not file:
        return jsonify({"ok": False, "msg": "arquivo 'foto' não enviado"}), 400

    # valida extensão
    original = (file.filename or "").strip()
    ext = original.rsplit(".", 1)[-1].lower() if "." in original else ""
    if ext not in ALLOWED_EXT:
        return jsonify({"ok": False, "msg": "Extensão inválida. Use jpg/jpeg/png"}), 400

    # limite simples de tamanho (para evitar travar o servidor)
    file.stream.seek(0, os.SEEK_END)
    size = file.stream.tell()
    file.stream.seek(0)
    if size > (MAX_MB * 1024 * 1024):
        return jsonify({"ok": False, "msg": f"Arquivo muito grande (> {MAX_MB}MB)"}), 400

    try:
        ped_id_int = int(ped_id)
    except Exception:
        return jsonify({"ok": False, "msg": "ped_id inválido"}), 400

    # salva com nome fixo {PED_ID}.jpg (padronizado)
    filename = secure_filename(f"{ped_id_int}.jpg")
    abs_path = os.path.join(PASTA_PEDIDOS, filename)

    # salva arquivo
    file.save(abs_path)

    # grava caminho relativo no banco
    rel_path = f"pedidos/{ped_id_int}.jpg"
    try:
        atualizar_foto_path(ped_id_int, rel_path)
    except Exception as e:
        # se der erro no banco, apaga o arquivo para não ficar lixo
        try:
            if os.path.exists(abs_path):
                os.remove(abs_path)
        except Exception:
            pass
        print("Erro ao atualizar foto path:", e)
        return jsonify({"ok": False, "msg": "Erro ao gravar caminho da foto no banco"}), 500

    return jsonify({"ok": True, "path": rel_path})

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import '../services/pedido_local_service.dart';

class EditarQuantidadePage extends StatefulWidget {
  final Map<String, dynamic> item;
  final double precoUnitario;
  final String cliente;
  final int pedidoId;

  const EditarQuantidadePage({
    super.key,
    required this.item,
    required this.precoUnitario,
    required this.cliente,
    required this.pedidoId,
  });

  @override
  State<EditarQuantidadePage> createState() => _EditarQuantidadePageState();
}

class _EditarQuantidadePageState extends State<EditarQuantidadePage> {
  late TextEditingController quantidadeCtrl;
  late TextEditingController recheio1Ctrl;
  late TextEditingController recheio2Ctrl;

  double subtotal = 0.0;

  @override
  void initState() {
    super.initState();

    quantidadeCtrl = TextEditingController(
      text: (widget.item['quantidade'] ?? 0).toString(),
    );

    // ✅ CARREGA RECHEIOS JÁ EXISTENTES
    recheio1Ctrl = TextEditingController(
      text: widget.item['descricao2']?.toString() ?? '',
    );

    recheio2Ctrl = TextEditingController(
      text: widget.item['descricao3']?.toString() ?? '',
    );

    _recalcularSubtotal();
  }

  // 🔍 BUSCA DE RECHEIOS (pode virar API / Firebird depois)
  Future<List<String>> _buscarRecheios(String filtro) async {
    final todos = <String>[
      'DOCE DE LEITE',
      'CHOCOLATE',
      'CHOCOLATE BRANCO',
      'NINHO',
      'COCO',
      'ABACAXI',
      'MORANGO',
      'CREME',
      'NOZES',
    ];

    return todos
        .where((r) => r.toLowerCase().contains(filtro.toLowerCase()))
        .toList();
  }

  void _recalcularSubtotal() {
    final qtd =
        double.tryParse(quantidadeCtrl.text.replaceAll(',', '.')) ?? 0.0;
    setState(() {
      subtotal = qtd * widget.precoUnitario;
    });
  }

  Future<void> _salvar() async {
    final novaQuantidade =
        double.tryParse(quantidadeCtrl.text.replaceAll(',', '.')) ?? 0.0;

    if (novaQuantidade <= 0) {
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Quantidade inválida')));
      return;
    }

    final novoSubtotal = novaQuantidade * widget.precoUnitario;

    await PedidoLocalService().atualizarItemBoloQuantidade(
      itemId: widget.item['id'],
      novaQuantidade: novaQuantidade,
      novoSubtotal: novoSubtotal,
      descricao2: recheio1Ctrl.text.trim(),
      descricao3: recheio2Ctrl.text.trim(),
    );

    if (!mounted) return;

    Navigator.pop(context, {
      'novaQuantidade': novaQuantidade,
      'novoSubtotal': novoSubtotal,
    });
  }

  @override
  void dispose() {
    quantidadeCtrl.dispose();
    recheio1Ctrl.dispose();
    recheio2Ctrl.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Editar Quantidade')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              'Cliente: ${widget.cliente}',
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 4),
            Text(
              'Pedido: ${widget.pedidoId}',
              style: const TextStyle(color: Colors.black54),
            ),
            const SizedBox(height: 16),

            // PRODUTO
            Text(
              widget.item['descricao1']?.toString() ?? '',
              style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
            ),

            const SizedBox(height: 16),

            // PREÇO
            Text(
              'Preço por KG: R\$ ${widget.precoUnitario.toStringAsFixed(2)}',
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),

            const SizedBox(height: 12),

            // QUANTIDADE
            TextField(
              controller: quantidadeCtrl,
              keyboardType: const TextInputType.numberWithOptions(
                decimal: true,
              ),
              inputFormatters: [
                FilteringTextInputFormatter.allow(
                  RegExp(r'^\d*([.,]\d{0,3})?$'),
                ),
              ],
              decoration: const InputDecoration(
                labelText: 'Quantidade (KG)',
                border: OutlineInputBorder(),
              ),
              onChanged: (_) => _recalcularSubtotal(),
            ),

            const SizedBox(height: 16),

            // =========================
            // RECHEIO 1 (AUTOCOMPLETE)
            // =========================
            Autocomplete<String>(
              initialValue: TextEditingValue(text: recheio1Ctrl.text),
              optionsBuilder: (textEditingValue) async {
                if (textEditingValue.text.isEmpty) {
                  return const Iterable<String>.empty();
                }
                return await _buscarRecheios(textEditingValue.text);
              },
              onSelected: (selection) {
                recheio1Ctrl.text = selection;
              },
              fieldViewBuilder:
                  (context, controller, focusNode, onFieldSubmitted) {
                    recheio1Ctrl = controller;
                    return TextField(
                      controller: controller,
                      focusNode: focusNode,
                      decoration: const InputDecoration(
                        labelText: 'Recheio 1',
                        border: OutlineInputBorder(),
                      ),
                    );
                  },
            ),

            const SizedBox(height: 12),

            // =========================

def round_money(value):
    try:
        return round(float(value), 2)

def parse_float(value, default=0.0):

def parse_date(value):
    """
    Aceita:
      - 'YYYY-MM-DD' (padrão comum do Flutter)
      - 'DD/MM/YYYY' (padrão BR)
      - date (já convertido)
    """
    if value is None:
        return None
    if isinstance(value, date):
        return value

    s = str(value).strip()
    if not s:
        return None

    for fmt in ("%Y-%m-%d", "%d/%m/%Y"):
        try:
            return datetime.strptime(s, fmt).date()
        except Exception:
            pass

    # Se não conseguir converter, retorna como string mesmo (Firebird às vezes aceita),
    # mas é melhor mandar no formato 'YYYY-MM-DD' do app.
    return s


def parse_time(value):
    """
    Aceita:
      - 'HH:MM'
      - 'HH:MM:SS'
      - 'HH:MM AM/PM' ou 'HH:MM PM'
      - time (já convertido)
    """
    if value is None:
        return None
    if isinstance(value, time):
        return value

    s = str(value).strip()
    if not s:
        return None

    for fmt in ("%H:%M", "%H:%M:%S", "%I:%M %p"):
        try:
            return datetime.strptime(s, fmt).time()
        except Exception:
            pass

    return None

class CarrinhoService {
  // Singleton
  static final CarrinhoService _instance = CarrinhoService._internal();
  factory CarrinhoService() => _instance;
  CarrinhoService._internal();

  // Lista de itens do carrinho
  final List<Map<String, dynamic>> _itens = [];

  // =========================
  // GETTERS
  // =========================

  List<Map<String, dynamic>> get itens => _itens;

  bool get vazio => _itens.isEmpty;

  double get total {
    double soma = 0.0;
    for (final item in _itens) {
      final subtotal = item['subtotal'] ?? 0.0;
      soma += subtotal;
    }
    return _round2(soma);
  }

  // =========================
  // AÇÕES
  // =========================

  void adicionarItem(Map<String, dynamic> item) {
    // 🔐 Blindagem monetária
    final quantidade = item['quantidade'] ?? 0.0;
    final precoUnitario = item['preco_unitario'] ?? item['preco'] ?? 0.0;

    final subtotal = _round2(quantidade * precoUnitario);

    item['preco_unitario'] = _round2(precoUnitario);
    item['subtotal'] = subtotal;

    _itens.add(item);
  }

  void removerItem(int index) {
    if (index >= 0 && index < _itens.length) {
      _itens.removeAt(index);
    }
  }

  void limpar() {
    _itens.clear();
  }

  // =========================
  // UTIL
  // =========================

  double _round2(double value) {
    return double.parse(value.toStringAsFixed(2));
  }
}

            f"{float(v):,.2f}"
            .replace(",", "X")
            .replace(".", ",")
            .replace("X", ".")
        )
    except Exception:
        return "0,00"

def parse_date(value):
    if not value:
        return None
    if isinstance(value, date):
        return value
    for fmt in ("%Y-%m-%d", "%d/%m/%Y"):
        try:
            return datetime.strptime(str(value), fmt).date()
        except Exception:
            pass
    return None

def parse_time(value):
    if not value:
        return None
    if isinstance(value, time):
        return value
    for fmt in ("%H:%M", "%H:%M:%S"):
        try:
            return datetime.strptime(str(value), fmt).time()
        except Exception:
            pass
    return None

                PED_PAGAMENTO,
                PED_DATA_ENT,
                PED_HORA_ENT,

            pedido.get("contato"),
            pedido.get("forma_pagamento"),
            parse_date(pedido.get("data_entrega")),

  Future<Database> _initDb() async {
    final path = join(await getDatabasesPath(), 'padaria.db');

    return await openDatabase(
      path,
      version: 3, // ⬅️ versão incrementada
      onCreate: (db, version) async {
        await db.execute('''
        CREATE TABLE pedido (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          numero_pedido INTEGER,
          cliente TEXT,
          contato TEXT,
          forma_pagamento TEXT,
          data_entrega TEXT,
          hora_entrega TEXT,
          observacoes TEXT,
          foto_bolo_local TEXT,
          total REAL,
          status TEXT,
          criado_em TEXT
        )
      ''');

        await db.execute('''
        CREATE TABLE pedido_item (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          pedido_id INTEGER,
          item_tipo TEXT,
          descricao1 TEXT,
          descricao2 TEXT,
          descricao3 TEXT,
          unidade TEXT,
          quantidade REAL,
          preco_unitario REAL,
          subtotal REAL,
          rec_id INTEGER,
          observacao TEXT
        )
      ''');
      },
      onUpgrade: (db, oldVersion, newVersion) async {
        if (oldVersion < 2) {
          await db.execute(
            "ALTER TABLE pedido ADD COLUMN foto_bolo_local TEXT",
          );
        }

        if (oldVersion < 3) {
          await db.execute(
            "ALTER TABLE pedido ADD COLUMN numero_pedido INTEGER",
          );
        }
      },
    );
  }

  // =========================
  // UTIL
  // =========================

  double _round2(double value) {
    return double.parse(value.toStringAsFixed(2));
  }

  // =========================
  // SALVAR PEDIDO
  // =========================

  Future<int> salvarPedido({
    required Map<String, dynamic> cabecalho,
    required List<Map<String, dynamic>> itens,
  }) async {
    final db = await database;

    // 🔢 Numeração segura (SharedPreferences)
    final numeroPedido = await NumeracaoPedidoService.proximoNumero();

    // 🔐 Blindagem do total
    final total = _round2((cabecalho['total'] ?? 0.0).toDouble());

    // 🧾 Insere pedido
    final pedidoId = await db.insert('pedido', {
      ...cabecalho,
      'numero_pedido': numeroPedido,
      'total': total,
      'status': 'ENVIAR',
      'criado_em': DateTime.now().toIso8601String(),
    });

    // 🔐 Blindagem item a item
    for (final item in itens) {
      final quantidade = (item['quantidade'] ?? 0).toDouble();
      final precoUnitario = (item['preco_unitario'] ?? 0).toDouble();

      final subtotal = _round2(quantidade * precoUnitario);

      await db.insert('pedido_item', {
        'pedido_id': pedidoId,
        'item_tipo': item['item_tipo'],
        'descricao1': item['descricao1'],
        'descricao2': item['descricao2'],
        'descricao3': item['descricao3'],
        'unidade': item['unidade'],
        'quantidade': quantidade,
        'preco_unitario': _round2(precoUnitario),
        'subtotal': subtotal,
        'rec_id': item['rec_id'],
        'observacao': item['observacao'],
      });
    }

    return pedidoId;
  }

  Future<List<Map<String, dynamic>>> listarItensDoPedido(int pedidoId) async {
    final db = await database;

    final itens = await db.query(
      'pedido_item',
      where: 'pedido_id = ?',
      whereArgs: [pedidoId],
    );

    double parseMoney(String? v) {
      if (v == null) return 0.0;
      var s = v.trim();
      if (s.isEmpty) return 0.0;

      s = s.replaceAll('R\$', '').replaceAll(' ', '');

      if (s.contains('.') && s.contains(',')) {
        final lastDot = s.lastIndexOf('.');
        final lastComma = s.lastIndexOf(',');
        if (lastComma > lastDot) {
          // 1.234,56 => pt_BR
          s = s.replaceAll('.', '').replaceAll(',', '.');
          return double.tryParse(s) ?? 0.0;
        } else {
          // 1,234.56 => en_US
          s = s.replaceAll(',', '');
          return double.tryParse(s) ?? 0.0;
        }
      }

      if (s.contains(',')) {
        // 1.234,56 => remove milhar e troca decimal
        s = s.replaceAll('.', '').replaceAll(',', '.');
        return double.tryParse(s) ?? 0.0;
      }

      return double.tryParse(s) ?? 0.0;
    }

    String getCab(Map<String, String> c, List<String> keys) {
      for (final k in keys) {
        final v = c[k];
        if (v != null && v.trim().isNotEmpty) return v.trim();
      }

      for (final e in c.entries) {
        final kk = e.key.trim().toLowerCase();
        for (final wanted in keys) {
          final w = wanted.trim().toLowerCase();
          if (w.isEmpty) continue;
          if (kk == w || kk.contains(w)) {
            final vv = e.value.trim();
            if (vv.isNotEmpty) return vv;
          }
        }
      }

      return '';

    /// Se vier "ENTREGA: 2025-12-29 20:24" ou "2025-12-29T20:24", separa data/hora.
    /// E converte data ISO -> BR (29/12/2025).
    Map<String, String> splitEntregaIfNeeded(
      String dataEntrega,
      String horaEntrega,
    ) {
      var d = dataEntrega.trim();
      var h = horaEntrega.trim();

      if (h.isEmpty) {
        final m = RegExp(r'^(.+?)[T ](\d{1,2}:\d{2}(:\d{2})?)$').firstMatch(d);
        if (m != null) {
          d = m.group(1)!.trim();
          h = m.group(2)!.trim();
        }
      }

      // ISO yyyy-mm-dd -> dd/mm/yyyy
      final iso = RegExp(r'^(\d{4})-(\d{2})-(\d{2})').firstMatch(d);
      if (iso != null) {
        final yyyy = iso.group(1)!;
        final mm = iso.group(2)!;
        final dd = iso.group(3)!;
        d = '$dd/$mm/$yyyy';
      }

      return {'data': d, 'hora': h};
    }

class FirebirdService {
  //static const String baseUrl = 'http://192.168.1.238:5002';
  //static const String baseUrl = 'http://192.168.0.250:5002';
  //static const String baseUrl = 'http://192.168.1.174:5002';
  static Future<bool> testarConexao() async {
    final url = Uri.parse('${ApiService.baseUrl}/ping');
    final resp = await http.get(url);
    return resp.statusCode == 200;
  }

  // ==================================================
  // ENVIO FINAL DO PEDIDO (LÊ SQLITE → ENVIA API)
  // ==================================================
  static Future<void> enviarPedidoPorId(int pedidoId) async {
    final pedidoLocal = PedidoLocalService();

    // 1️⃣ Busca cabeçalho
    final pedidos = await pedidoLocal.listarPedidosPendentes();
    final pedido = pedidos.firstWhere(
      (p) => p['id'] == pedidoId,
      orElse: () => throw Exception('Pedido não encontrado'),
    );

    // 2️⃣ Busca itens
    final itens = await pedidoLocal.listarItensDoPedido(pedidoId);

    // 3️⃣ Monta JSON FINAL (AGORA COMPLETO)
    final payload = {
      'pedido': {
        'cliente': pedido['cliente'],
        'contato': pedido['contato'],
        'forma_pagamento': pedido['forma_pagamento'],
        'data_entrega': pedido['data_entrega'],
        'hora_entrega': pedido['hora_entrega'],
        'observacoes': pedido['observacoes'],
        'total': pedido['total'],
      },
      'itens': itens.map((item) {
        return {
          'item_tipo': item['item_tipo'], // P ou B
          'descricao1': item['descricao1'],
          'descricao2': item['descricao2'],
          'descricao3': item['descricao3'],
          'unidade': item['unidade'],
          'quantidade': item['quantidade'],
          'preco_unitario': item['preco_unitario'],
          'subtotal': item['subtotal'],

          // 🔴 CAMPOS QUE FALTAVAM (BOLO)
          'rec_id': item['rec_id'], // ID DO RECHEIO
          'observacao': item['observacao'], // OBS DO ITEM
        };
      }).toList(),
    };

    // 4️⃣ Envia para API
    final response = await http.post(
      Uri.parse('${ApiService.baseUrl}/pedidos'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode(payload),
    );

    if (response.statusCode != 200) {
      throw Exception('Erro ao enviar pedido: ${response.body}');
    }

    // 4️⃣1️⃣ Captura o PED_ID gerado no servidor
    final respJson = jsonDecode(response.body);
    final pedIdServidor = respJson['pedido_id'];

    // 4️⃣2️⃣ Upload da foto do bolo (se existir)
    final fotoPath = pedido['foto_bolo_local'];
    if (pedIdServidor != null &&
        fotoPath != null &&
        fotoPath.toString().isNotEmpty) {
      try {
        await ApiService.uploadFotoPedido(
          pedidoId: int.parse(pedIdServidor.toString()),
          fotoPathLocal: fotoPath.toString(),
        );
      } catch (e) {
        // Pedido já foi enviado, então só registra (não perde o pedido)
        print('⚠️ Pedido enviado, mas falhou o upload da foto: $e');
      }
    }

    // 5️⃣ Atualiza status local
    await pedidoLocal.atualizarStatusPedido(pedidoId, 'ENVIADO');
  }
}

import 'dart:io';

import '../services/pedido_local_service.dart';
import '../services/firebird_service.dart';
import '../services/pedido_pdf_service.dart';

class PedidosPendentesPage extends StatefulWidget {
  const PedidosPendentesPage({super.key});

  @override
  State<PedidosPendentesPage> createState() => _PedidosPendentesPageState();
}

class _PedidosPendentesPageState extends State<PedidosPendentesPage> {
  late Future<List<Map<String, dynamic>>> _pedidosFuture;
  bool _enviando = false;

  @override
  void initState() {
    super.initState();
    _carregarPedidos();
  }

  void _carregarPedidos() {
    _pedidosFuture = PedidoLocalService().listarPedidosPendentes();
  }

  // =========================
  // UTIL
  // =========================
  double _toDouble(dynamic v) {
    if (v == null) return 0.0;
    if (v is num) return v.toDouble();
    if (v is String) {
      final s = v.trim().replaceAll(' ', '').replaceAll(',', '.');
      return double.tryParse(s) ?? 0.0;
    }
    return 0.0;
  }

  String _montarDescricaoItem(Map<String, dynamic> it) {
    final p1 = (it['descricao1'] ?? '').toString().trim();
    final p2 = (it['descricao2'] ?? '').toString().trim();
    final p3 = (it['descricao3'] ?? '').toString().trim();
    return [p1, p2, p3].where((s) => s.isNotEmpty).join(' ');
  }

  String _formatQtd(double q) {
    if (q == q.roundToDouble()) return q.toInt().toString();
    return q.toString();
  }

  // =========================
  // ENVIAR / REENVIAR PEDIDO
  // =========================
  Future<void> _enviarPedido(Map<String, dynamic> pedido) async {
    if (_enviando) return;

    setState(() => _enviando = true);
    final int pedidoId = (pedido['id'] as num).toInt();

    try {
      await FirebirdService.enviarPedidoPorId(pedidoId);
      await PedidoLocalService().atualizarStatusPedido(pedidoId, 'FATURADO');

      if (!mounted) return;
      setState(_carregarPedidos);

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Pedido faturado com sucesso')),
      );
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text('Erro ao enviar pedido: $e')));
    } finally {
      if (mounted) setState(() => _enviando = false);
    }
  }

  // =========================
  // GERAR ARQUIVO PDF (retorna File)
  // =========================
  Future<File> _gerarArquivoPdfDoPedido(Map<String, dynamic> pedido) async {
    final cabecalho = <String, String>{
      'Cliente': (pedido['cliente'] ?? '').toString(),
      'Contato': (pedido['contato'] ?? '').toString(),
      'Pagamento': (pedido['forma_pagamento'] ?? '').toString(),
      'Entrega':
          '${(pedido['data_entrega'] ?? '').toString()} ${(pedido['hora_entrega'] ?? '').toString()}',
      'Observações': (pedido['observacoes'] ?? '-').toString(),
      'foto_bolo_local': (pedido['foto_bolo_local'] ?? '').toString(),
    };

    final int pedidoId = (pedido['id'] as num).toInt();
    final itensDb = await PedidoLocalService().listarItensDoPedido(pedidoId);

    final itensProdutos = <Map<String, String>>[];
    final itensBolos = <Map<String, String>>[];

    for (final it in itensDb) {
      final tipo = (it['item_tipo'] ?? '').toString().trim().toUpperCase();

      final qtdNum = _toDouble(it['quantidade']);
      final unit = _toDouble(it['preco_unitario']);
      final sub = _toDouble(it['subtotal']);

      final desc1 = (it['descricao1'] ?? '').toString().trim();
      final desc2 = (it['descricao2'] ?? '').toString().trim();
      final desc3 = (it['descricao3'] ?? '').toString().trim();
      final obs = (it['observacao'] ?? '').toString().trim();

      // ✅ Regra: bolo NÃO concatena recheios na "desc"
      final descParaPdf = (tipo == 'B') ? desc1 : _montarDescricaoItem(it);

      final map = <String, String>{
        // Compatibilidade + regra do PDF
        'ITEM_TIPO': tipo,

        // ✅ manda separado (principal correção)
        'ITEM_DESC1': desc1,
        'ITEM_DESC2': desc2,
        'ITEM_DESC3': desc3,

        // ✅ OBS (mantém os dois nomes pra não quebrar nada)
        'ITEM_OBS': obs,
        'obs': obs,

        // ✅ o PDF antigo podia ler "desc", então mantemos
        'desc': descParaPdf,

        'qtd': _formatQtd(qtdNum),
        'unit': unit.toStringAsFixed(2),
        'sub': sub.toStringAsFixed(2),
      };

      if (tipo == 'B') {
        itensBolos.add(map);
      } else {
        itensProdutos.add(map);
      }
    }

    //print('ANTES DO PDF - foto_bolo_local: ${cabecalho['foto_bolo_local']}');
    //print('ANTES DO PDF - CABECALHO COMPLETO: $cabecalho');

    return PedidoPdfService.gerarPedidoPdf(
      titulo: 'Pedido #$pedidoId',
      cabecalho: cabecalho,
      itensProdutos: itensProdutos,
      itensBolos: itensBolos,
    );
  }

  // =========================
  // VER / IMPRIMIR PDF
  // =========================
  Future<void> _verPdf(Map<String, dynamic> pedido) async {
    try {
      final file = await _gerarArquivoPdfDoPedido(pedido);
      final Uint8List bytes = await file.readAsBytes();

  }

  // =========================

  Color _corTextoBotao(String status) {

    }
  }

  }

      appBar: AppBar(title: const Text('Pedidos')),

          final pedidos = snapshot.data ?? [];
          if (pedidos.isEmpty) {
            return const Center(child: Text('Nenhum pedido encontrado'));
          }

          return ListView.builder(
            itemCount: pedidos.length,
            itemBuilder: (context, index) {
              final pedido = pedidos[index];

              final String status = (pedido['status'] == 'PENDENTE')
                  ? 'ENVIAR'
                  : (pedido['status'] ?? '').toString();

              final bool isEnviar = status == 'ENVIAR';

              return Card(

    double parseMoney(String? v) {
      if (v == null) return 0.0;
      var s = v.trim();
      if (s.isEmpty) return 0.0;

      s = s.replaceAll('R\$', '').replaceAll(' ', '');

      if (s.contains('.') && s.contains(',')) {
        final lastDot = s.lastIndexOf('.');
        final lastComma = s.lastIndexOf(',');
        if (lastComma > lastDot) {
          // 1.234,56 => pt_BR
          s = s.replaceAll('.', '').replaceAll(',', '.');
          return double.tryParse(s) ?? 0.0;
        } else {
          // 1,234.56 => en_US
          s = s.replaceAll(',', '');
          return double.tryParse(s) ?? 0.0;
        }
      }

      if (s.contains(',')) {
        // 1.234,56 => remove milhar e troca decimal
        s = s.replaceAll('.', '').replaceAll(',', '.');
        return double.tryParse(s) ?? 0.0;
      }

      return double.tryParse(s) ?? 0.0;
    }

    String getCab(Map<String, String> c, List<String> keys) {
      for (final k in keys) {
        final v = c[k];
        if (v != null && v.trim().isNotEmpty) return v.trim();
      }

      for (final e in c.entries) {
        final kk = e.key.trim().toLowerCase();
        for (final wanted in keys) {
          final w = wanted.trim().toLowerCase();
          if (w.isEmpty) continue;
          if (kk == w || kk.contains(w)) {

    /// Se vier "ENTREGA: 2025-12-29 20:24" ou "2025-12-29T20:24", separa data/hora.
    /// E converte data ISO -> BR (29/12/2025).
    Map<String, String> splitEntregaIfNeeded(
      String dataEntrega,
      String horaEntrega,
    ) {
      var d = dataEntrega.trim();
      var h = horaEntrega.trim();

      if (h.isEmpty) {
        final m = RegExp(r'^(.+?)[T ](\d{1,2}:\d{2}(:\d{2})?)$').firstMatch(d);
        if (m != null) {
          d = m.group(1)!.trim();
          h = m.group(2)!.trim();
        }
      }

      // ISO yyyy-mm-dd -> dd/mm/yyyy
      final iso = RegExp(r'^(\d{4})-(\d{2})-(\d{2})').firstMatch(d);
      if (iso != null) {
        final yyyy = iso.group(1)!;
        final mm = iso.group(2)!;
        final dd = iso.group(3)!;
        d = '$dd/$mm/$yyyy';
      }

      return {'data': d, 'hora': h};
    }

    // =========================
    // MONEY (pt_BR + robust parse)
    // =========================
    double parseMoney(String? v) {
      if (v == null) return 0.0;
      var s = v.trim();
      if (s.isEmpty) return 0.0;

      s = s.replaceAll('R\$', '').replaceAll(' ', '');

      if (s.contains('.') && s.contains(',')) {
        final lastDot = s.lastIndexOf('.');
        final lastComma = s.lastIndexOf(',');
        if (lastComma > lastDot) {
          // 1.234,56 => pt_BR
          s = s.replaceAll('.', '').replaceAll(',', '.');
          return double.tryParse(s) ?? 0.0;
        } else {
          // 1,234.56 => en_US
          s = s.replaceAll(',', '');
          return double.tryParse(s) ?? 0.0;
        }
      }

      if (s.contains(',')) {
        // 1.234,56 => remove milhar e troca decimal
        s = s.replaceAll('.', '').replaceAll(',', '.');
        return double.tryParse(s) ?? 0.0;
      }

      return double.tryParse(s) ?? 0.0;
    }

    final fmtMoney = NumberFormat.currency(
      locale: 'pt_BR',
      symbol: 'R\$ ',
      decimalDigits: 2,
    );

    String money(num v) => fmtMoney.format(v);
    String moneyFromRaw(String? raw) => money(parseMoney(raw));

    double sumSubtotal(List<Map<String, String>> itens) {
      double total = 0.0;
      for (final it in itens) {
        final raw = it['sub'] ?? it['subtotal'] ?? '0';

      }
      return total;

    String getCab(Map<String, String> c, List<String> keys) {
      for (final k in keys) {
        final v = c[k];
        if (v != null && v.trim().isNotEmpty) return v.trim();
      }

      for (final e in c.entries) {
        final kk = e.key.trim().toLowerCase();
        for (final wanted in keys) {

          if (w.isEmpty) continue;
          if (kk == w || kk.contains(w)) {

            if (vv.isNotEmpty) return vv;

        }

    ) {
      var d = dataEntrega.trim();
      var h = horaEntrega.trim();

      if (h.isEmpty) {
        final m = RegExp(r'^(.+?)[T ](\d{1,2}:\d{2}(:\d{2})?)$').firstMatch(d);
        if (m != null) {
          d = m.group(1)!.trim();
          h = m.group(2)!.trim();
        }
      }

      // ISO yyyy-mm-dd -> dd/mm/yyyy
      final iso = RegExp(r'^(\d{4})-(\d{2})-(\d{2})').firstMatch(d);
      if (iso != null) {
        final yyyy = iso.group(1)!;
        final mm = iso.group(2)!;
        final dd = iso.group(3)!;

  double _toDouble(dynamic v) {
    if (v == null) return 0.0;
    if (v is num) return v.toDouble();
    if (v is String) {
      final s = v.trim().replaceAll(' ', '').replaceAll(',', '.');
      return double.tryParse(s) ?? 0.0;
    }
    return 0.0;
  }

    final p2 = (it['descricao2'] ?? '').toString().trim();

    return [p1, p2, p3].where((s) => s.isNotEmpty).join(' ');
  }

          '${(pedido['data_entrega'] ?? '').toString()} ${(pedido['hora_entrega'] ?? '').toString()}',
      'Observações': (pedido['observacoes'] ?? '-').toString(),
    };

    final int pedidoId = (pedido['id'] as num).toInt();
    final itensDb = await PedidoLocalService().listarItensDoPedido(pedidoId);

    final itensProdutos = <Map<String, String>>[];
    final itensBolos = <Map<String, String>>[];

    for (final it in itensDb) {
      final tipo = (it['item_tipo'] ?? '').toString().trim().toUpperCase();

      final qtdNum = _toDouble(it['quantidade']);
      final unit = _toDouble(it['preco_unitario']);
      final sub = _toDouble(it['subtotal']);

      final desc1 = (it['descricao1'] ?? '').toString().trim();
      final desc2 = (it['descricao2'] ?? '').toString().trim();
      final desc3 = (it['descricao3'] ?? '').toString().trim();
      final obs = (it['observacao'] ?? '').toString().trim();

      // ✅ Regra: bolo NÃO concatena recheios na "desc"
      final descParaPdf = (tipo == 'B') ? desc1 : _montarDescricaoItem(it);

      final map = <String, String>{
        // Compatibilidade + regra do PDF
        'ITEM_TIPO': tipo,

        // ✅ manda separado (principal correção)
        'ITEM_DESC1': desc1,
        'ITEM_DESC2': desc2,
        'ITEM_DESC3': desc3,

        // ✅ OBS (mantém os dois nomes pra não quebrar nada)
        'ITEM_OBS': obs,
        'obs': obs,

        // ✅ o PDF antigo podia ler "desc", então mantemos
        'desc': descParaPdf,

        'qtd': _formatQtd(qtdNum),
        'unit': unit.toStringAsFixed(2),
        'sub': sub.toStringAsFixed(2),
      };

      if (tipo == 'B') {
        itensBolos.add(map);
      } else {
        itensProdutos.add(map);
      }
    }

    return PedidoPdfService.gerarPedidoPdf(
      titulo: 'Pedido #$pedidoId',
      cabecalho: cabecalho,

  Future<Database> _initDb() async {
    final path = join(await getDatabasesPath(), 'padaria.db');

    return await openDatabase(
      path,
      version: 1,
      onCreate: (db, version) async {
        await db.execute('''
          CREATE TABLE pedido (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            cliente TEXT,
            contato TEXT,
            forma_pagamento TEXT,
            data_entrega TEXT,
            hora_entrega TEXT,
            observacoes TEXT,
            total REAL,
            status TEXT,
            criado_em TEXT
          )
        ''');

        await db.execute('''
  CREATE TABLE pedido_item (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pedido_id INTEGER,
    item_tipo TEXT,        -- 👈 NOVO
    descricao1 TEXT,
    descricao2 TEXT,
    descricao3 TEXT,
    unidade TEXT,
    quantidade REAL,
    preco_unitario REAL,
    subtotal REAL,
    rec_id INTEGER,        -- 👈 NOVO
    observacao TEXT        -- 👈 NOVO
  )
''');
      },
    );
  }

  // =========================
  // UTIL
  // =========================

  double _round2(double value) {
    return double.parse(value.toStringAsFixed(2));
  }

  // =========================
  // SALVAR PEDIDO
  // =========================

  Future<int> salvarPedido({
    required Map<String, dynamic> cabecalho,
    required List<Map<String, dynamic>> itens,
  }) async {
    final db = await database;

    // 🔐 Blindagem do total
    final total = _round2((cabecalho['total'] ?? 0.0).toDouble());

    final pedidoId = await db.insert('pedido', {
      ...cabecalho,
      'total': total,
      'status': 'ENVIAR',
      'criado_em': DateTime.now().toIso8601String(),
    });

    // 🔐 Blindagem item a item
    for (final item in itens) {
      final quantidade = (item['quantidade'] ?? 0).toDouble();
      final precoUnitario = (item['preco_unitario'] ?? 0).toDouble();

      final subtotal = _round2(quantidade * precoUnitario);

      await db.insert('pedido_item', {
        'pedido_id': pedidoId,
        'item_tipo': item['item_tipo'], // 👈 AGORA SALVA
        'descricao1': item['descricao1'],
        'descricao2': item['descricao2'],
        'descricao3': item['descricao3'],
        'unidade': item['unidade'],
        'quantidade': quantidade,
        'preco_unitario': _round2(precoUnitario),
        'subtotal': subtotal,
        'rec_id': item['rec_id'], // 👈 BOLO
        'observacao': item['observacao'], // 👈 BOLO
      });
    }

    return pedidoId;
  }

class _CabecalhoPageState extends State<CabecalhoPage> {
  final ImagePicker _picker = ImagePicker();

  String? _fotoPath;

  // =========================
  // CAPTURAR / ESCOLHER FOTO
  // =========================
  Future<void> _selecionarFoto(ImageSource source) async {
    final XFile? image = await _picker.pickImage(
      source: source,
      imageQuality: 80,
    );

    if (image == null) return;

    final dir = await getApplicationDocumentsDirectory();
    final nomeArquivo = 'pedido_${DateTime.now().millisecondsSinceEpoch}.jpg';
    final destino = File('${dir.path}/$nomeArquivo');

    await File(image.path).copy(destino.path);

    setState(() {
      _fotoPath = destino.path;
    });

    _emitirCabecalho();
  }

  // =========================
  // EMITE CABECALHO COMPLETO
  // =========================
  void _emitirCabecalho() {
    final dataEntregaStr = widget.dataEntrega != null
        ? DateFormat('dd/MM/yyyy').format(widget.dataEntrega!)
        : '';

    final horaEntregaStr = widget.horaEntrega != null
        ? widget.horaEntrega!.format(context)
        : '';

    widget.onChanged({
      'cliente': widget.clienteCtrl.text,
      'contato': widget.contatoCtrl.text,
      'forma_pagamento': widget.formaPagamento,
      'data_entrega': dataEntregaStr,
      'hora_entrega': horaEntregaStr,
      'observacoes': widget.observacaoCtrl.text,
      'foto_bolo_local': _fotoPath ?? '',
    });
  }

  // =========================
  // UI
  // =========================
  @override
  Widget build(BuildContext context) {
    return SingleChildScrollView(
      padding: const EdgeInsets.all(16),
      child: Column(
        children: [
          _campoTexto('Cliente', widget.clienteCtrl),
