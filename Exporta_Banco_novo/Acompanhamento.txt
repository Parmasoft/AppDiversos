    def _adicionar_campo_inline(self, parent, titulo, column_start, valor="", largura=12, show=None):
        ttk.Label(parent, text=titulo, style="Campo.TLabel").grid(
            row=0, column=column_start, sticky="w", padx=(0, 6)
        )

        widget = ttk.Entry(parent, width=largura)
        if show:
            widget.configure(show=show)

        if valor:
            widget.insert(0, valor)

        widget.grid(row=0, column=column_start + 1, sticky="w", padx=(0, 14), ipady=2)
        return widget

    def _selecionar_arquivo_banco(self, lado):
        caminho = filedialog.askopenfilename(
            title="Selecionar banco Firebird",
            filetypes=[
                ("Banco Firebird", "*.fdb *.gdb"),
                ("Banco Firebird", "*.FDB *.GDB"),
                ("Todos os arquivos", "*.*"),
            ],
        )

        if not caminho:
            return

        if lado == "origem":
            self.edt_origem_banco.delete(0, tk.END)
            self.edt_origem_banco.insert(0, caminho)
        else:
            self.edt_destino_banco.delete(0, tk.END)
            self.edt_destino_banco.insert(0, caminho)

    def _obter_dados_conexao(self, lado):
        widgets = self._obter_widgets_lado(lado)

        porta_txt = (widgets["porta"].get() or "").strip()
        porta = 3050
        if porta_txt:
            try:
                porta = int(porta_txt)
            except ValueError:
                raise ValueError(f"A porta da conexão {lado} está inválida.")

        return {
            "servidor": (widgets["servidor"].get() or "").strip() or "127.0.0.1",
            "porta": porta,
            "banco": (widgets["banco"].get() or "").strip(),
            "usuario": (widgets["usuario"].get() or "").strip() or "SYSDBA",
            "senha": widgets["senha"].get() or "",
            "charset": (widgets["charset"].get() or "").strip() or "WIN1252",
        }

    def _montar_texto_ambiente(self):
        return self.firebird_service.montar_texto_ambiente(self.ambiente_firebird)

    def _conectar_automatico(self, dados):
        return self.firebird_service.conectar_automatico(dados, self.ambiente_firebird)

    def _fechar_conexao(self, con):
        self.firebird_service.fechar_conexao(con)

    def _testar_conexao(self, lado):
        con = None
        try:
            self.ambiente_firebird = self.firebird_service.detectar_ambiente_firebird()
            if self.lbl_status_conexao is not None:
                self.lbl_status_conexao.configure(text=self._montar_texto_ambiente())

            dados = self._obter_dados_conexao(lado)
            con, info = self._conectar_automatico(dados)

            cur = con.cursor()
            cur.execute("SELECT CURRENT_TIMESTAMP FROM RDB$DATABASE")
            linha = cur.fetchone()

            detalhe_cliente = "auto sem DLL fixa"
            if info["cliente"] and info["cliente"].get("dll"):
                detalhe_cliente = f"FB {info['cliente'].get('versao')} -> {info['cliente'].get('dll')}"

            messagebox.showinfo(
                "Conexão OK",
                f"Conexão {lado} realizada com sucesso.\n\n"
                f"Modo: {info['modo']}\n"
                f"Cliente: {detalhe_cliente}\n"
                f"Banco: {dados['banco']}\n"
                f"Data/hora do servidor: {linha[0]}",
            )
        except Exception as e:
            messagebox.showerror("Erro de conexão", f"Falha ao conectar na {lado}.\n\n{e}")
        finally:
            self._fechar_conexao(con)

    def _validar_conexoes(self):
        mensagens = []

        if not self.firebird_service.has_fdb():
            mensagens.append("- fdb não está instalado.")
        if not self.firebird_service.has_firebird_driver():
            mensagens.append("- firebird-driver não está instalado.")

        try:
            dados_origem = self._obter_dados_conexao("origem")
            if not dados_origem["banco"]:
                mensagens.append("- Informe o caminho do banco de origem.")
        except Exception as e:
            mensagens.append(f"- Origem: {e}")

        try:
            dados_destino = self._obter_dados_conexao("destino")
            if not dados_destino["banco"]:
                mensagens.append("- Informe o caminho do banco de destino.")
        except Exception as e:
            mensagens.append(f"- Destino: {e}")

        if mensagens:
            messagebox.showwarning(
                "Validação",
                "Ajuste os pontos abaixo antes de seguir:\n\n" + "\n".join(mensagens),
            )
            return

        con_origem = None
        con_destino = None

        try:
            con_origem, _ = self._conectar_automatico(dados_origem)
            con_destino, _ = self._conectar_automatico(dados_destino)

            self._escrever_log("Conexão origem validada com sucesso.")
            self._escrever_log("Conexão destino validada com sucesso.")
            self.lbl_resumo_execucao.configure(
                text="Conexões válidas. Já é possível analisar e iniciar a migração."
            )

            messagebox.showinfo(
                "Validação",
                "Origem e destino conectados com sucesso.",
            )
        except Exception as e:
            messagebox.showerror(
                "Erro na validação",
                f"Não foi possível validar as conexões.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con_origem)
            self._fechar_conexao(con_destino)

        indice = selecionados[0]

    def _ao_selecionar_pacote(self, event=None):
        if self.lst_pacotes is None:
            return

        selecionados = self.lst_pacotes.curselection()

        if selecionados:
            indice = selecionados[0]

            if not self._pacote_esta_liberado(indice):
                self.lst_pacotes.selection_clear(0, tk.END)
                self.lst_pacotes.selection_set(0)
                self.lst_pacotes.activate(0)

                self._escrever_log(
                    "Pacote bloqueado. Processe primeiro os pacotes base antes do financeiro."
                )

                messagebox.showwarning(
                    "Pacote bloqueado",
                    "Processe primeiro os pacotes de Cadastros básicos, Clientes, Fornecedores e Produtos antes de migrar o financeiro em aberto.",
                )

        pacote = self._obter_pacote_selecionado()

        self.lst_tabelas.delete(0, tk.END)

        if not pacote:
            return

        for tabela in pacote["tabelas"]:
            self.lst_tabelas.insert(tk.END, tabela)

        self.lbl_resumo_execucao.configure(
            text=f"Pacote selecionado: {pacote['nome']} | {len(pacote['tabelas'])} tabela(s)."
        )

    def _obter_tabela_selecionada(self):
        if self.lst_tabelas is None:
            return None

        selecionados = self.lst_tabelas.curselection()
        if not selecionados:
            return None

        return self.lst_tabelas.get(selecionados[0])

    def _analisar_destino_pacote(self):
        pacote = self._obter_pacote_selecionado()
        if not pacote:
            messagebox.showwarning("Pacote", "Selecione um pacote.")
            return

        con_origem = None
        con_destino = None

        try:
            dados_origem = self._obter_dados_conexao("origem")
            dados_destino = self._obter_dados_conexao("destino")

            con_origem, _ = self._conectar_automatico(dados_origem)
            con_destino, _ = self._conectar_automatico(dados_destino)

            self._escrever_log("=" * 70)
            self._escrever_log(f"Análise do pacote: {pacote['nome']}")

            for tabela in pacote["tabelas"]:
                qtd_origem = self._contar_registros(con_origem, tabela)
                qtd_destino = self._contar_registros(con_destino, tabela)

                tabela_norm = self.firebird_service.normalizar_nome_tabela(tabela)

                if tabela_norm in self._obter_tabelas_financeiro_controlado():
                    sql_count_filtrado = self._montar_sql_count_origem_tabela(tabela_norm)
                    qtd_filtrada = 0

                    if sql_count_filtrado:
                        cur_filtro = con_origem.cursor()
                        cur_filtro.execute(sql_count_filtrado)
                        linha_filtro = cur_filtro.fetchone()
                        qtd_filtrada = int(linha_filtro[0] or 0)

                    self._escrever_log(
                        f"{tabela}: origem_total={qtd_origem} | origem_filtrada={qtd_filtrada} | destino={qtd_destino}"
                    )
                else:
                    self._escrever_log(
                        f"{tabela}: origem={qtd_origem} | destino={qtd_destino}"
                    )

            self._escrever_log("=" * 70)
            self.lbl_resumo_execucao.configure(
                text=f"Análise concluída para o pacote: {pacote['nome']}."
            )
        except Exception as e:
            messagebox.showerror(
                "Erro na análise",
                f"Não foi possível analisar o pacote selecionado.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con_origem)
            self._fechar_conexao(con_destino)

    def _contar_registros(self, con, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)
        cur = con.cursor()
        cur.execute(f"SELECT COUNT(*) FROM {tabela}")
        linha = cur.fetchone()
        return int(linha[0] or 0)

    def _registro_existe_destino(self, con_destino, tabela, campos_chave, dados):
        where = " AND ".join([f"{campo} = ?" for campo in campos_chave])
        valores = [dados[campo] for campo in campos_chave]

        cur = con_destino.cursor()
        cur.execute(
            f"SELECT FIRST 1 1 FROM {tabela} WHERE {where}",
            valores,
        )
        return cur.fetchone() is not None

    def _executar_update_destino(self, con_destino, tabela, dados, campos_chave):
        colunas_update = [col for col in dados.keys() if col not in campos_chave]

        if not colunas_update:
            return

        set_sql = ", ".join([f"{col} = ?" for col in colunas_update])
        where_sql = " AND ".join([f"{campo} = ?" for campo in campos_chave])

        sql = f"UPDATE {tabela} SET {set_sql} WHERE {where_sql}"

        valores = [dados[col] for col in colunas_update]
        valores.extend([dados[campo] for campo in campos_chave])

        cur = con_destino.cursor()
        cur.execute(sql, valores)
    def _limpar_valor_para_firebird(self, valor):
        if isinstance(valor, str):
            valor = valor.replace("\ufffd", "")
            valor = valor.replace("�", "")
            return valor

        return valor

    def _executar_insert_destino(self, con_destino, tabela, dados):
        if not dados:
            raise RuntimeError(f"Nenhum dado montado para inserir na tabela {tabela}.")

        colunas = list(dados.keys())
        placeholders = ", ".join(["?"] * len(colunas))
        sql = (
            f"INSERT INTO {tabela} ({', '.join(colunas)}) "
            f"VALUES ({placeholders})"
        )

        cur = con_destino.cursor()
        cur.execute(sql, [dados[col] for col in colunas])

    def _montar_dados_registro_por_colunas(self, registro, colunas_origem, colunas_destino):
        mapa_indices = {nome: i for i, nome in enumerate(colunas_origem)}
        colunas_comuns = [col for col in colunas_destino if col in mapa_indices]

        dados = {}
        for col in colunas_comuns:
            valor = registro[mapa_indices[col]]
            dados[col] = self._limpar_valor_para_firebird(valor)

        return dados

    def _migrar_lista_tabelas(self, tabelas, titulo_execucao):
        con_origem = None
        con_destino = None

        total_tabelas = 0
        sucesso_tabelas = 0
        falha_tabelas = 0
        total_lidos = 0
        total_inseridos = 0

        try:
            dados_origem = self._obter_dados_conexao("origem")
            dados_destino = self._obter_dados_conexao("destino")

            con_origem, _ = self._conectar_automatico(dados_origem)
            con_destino, _ = self._conectar_automatico(dados_destino)

            self._escrever_log("")
            self._escrever_log("=" * 70)
            self._escrever_log(f"Iniciando execução: {titulo_execucao}")
            self._escrever_log(f"Somente 10 registros: {'SIM' if self.var_somente_10.get() else 'NÃO'}")
            self._escrever_log(f"Limpar destino antes: {'SIM' if self.var_limpar_destino.get() else 'NÃO'}")
            self._escrever_log(f"Continuar em caso de erro: {'SIM' if self.var_continuar_erro.get() else 'NÃO'}")
            
            self._escrever_log("=" * 70)

            if self.pb_execucao is not None:
                self.pb_execucao.configure(maximum=1, value=0)

            if self.lbl_progresso_execucao is not None:
                self.lbl_progresso_execucao.configure(
                    text=f"Iniciando execução: {titulo_execucao}"
                )

            self.update_idletasks()

            for tabela in tabelas:
                total_tabelas += 1
                self.lbl_resumo_execucao.configure(text=f"Migrando tabela: {tabela} ...")

                if self.lbl_progresso_execucao is not None:
                    self.lbl_progresso_execucao.configure(
                        text=f"Iniciando tabela {total_tabelas}/{len(tabelas)}: {tabela}"
                    )

                if self.pb_execucao is not None:
                    total_registros_tabela = self._contar_registros_origem_migracao(
                        con_origem,
                        tabela,
                    )
                    self.pb_execucao.configure(
                        maximum=max(total_registros_tabela, 1),
                        value=0,
                    )

                self.update_idletasks()

                try:
                    lidos, inseridos = self._migrar_tabela(con_origem, con_destino, tabela)
                    total_lidos += lidos
                    total_inseridos += inseridos
                    sucesso_tabelas += 1
                    self._escrever_log(
                        f"[OK] {tabela} | lidos={lidos} | inseridos={inseridos}"
                    )
                except Exception as e:
                    falha_tabelas += 1
                    self._escrever_log(f"[ERRO] {tabela} | {e}")

                    try:
                        con_destino.rollback()
                    except Exception:
                        pass

                    if not self.var_continuar_erro.get():
                        raise RuntimeError(f"Execução interrompida na tabela {tabela}: {e}")

                finally:
                    self.update_idletasks()

            try:
                con_destino.commit()
            except Exception:
                pass

            resumo = (
                f"Execução concluída.\n\n"
                f"Tabelas processadas: {total_tabelas}\n"
                f"Tabelas com sucesso: {sucesso_tabelas}\n"
                f"Tabelas com falha: {falha_tabelas}\n"
                f"Registros lidos: {total_lidos}\n"
                f"Registros inseridos: {total_inseridos}"
            )

            self._escrever_log("-" * 70)
            self._escrever_log(resumo.replace("\n", " | "))
            self._escrever_log("=" * 70)

            
            self.lbl_resumo_execucao.configure(
                text=(
                    f"Concluído | sucesso={sucesso_tabelas} | "
                    f"falha={falha_tabelas} | inseridos={total_inseridos}"
                )
            )

            if self.lbl_progresso_execucao is not None:
                self.lbl_progresso_execucao.configure(
                    text=(
                        f"Concluído: {titulo_execucao} | "
                        f"sucesso={sucesso_tabelas} | falha={falha_tabelas}"
                    )
                )

            if titulo_execucao in ("Produtos", "Financeiro em aberto") and falha_tabelas == 0:
                self.base_inicial_processada = True
                self._atualizar_pacotes_liberados()
                self._escrever_log(
                    "Base estrutural marcada como processada. Financeiro em aberto liberado."
                )

            messagebox.showinfo("Migração concluída", resumo)



        except Exception as e:
            messagebox.showerror(
                "Erro na migração",
                f"Não foi possível concluir a migração.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con_origem)
            self._fechar_conexao(con_destino)

    def _montar_sql_origem_tabela(self, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        first = "FIRST 10 " if self.var_somente_10.get() else ""

        if tabela == "TB_RECEBIMENTO":
            return (
                f"SELECT {first}* "
                f"FROM TB_RECEBIMENTO "
                f"WHERE REC_STATUS IN ('AB', 'PB') "
                f"  AND COALESCE(REC_ABERTO, 'S') = 'S'"
            )

        if tabela == "TB_RECEBIMENTO_ITEM":
            return (
                f"SELECT {first}i.* "
                f"FROM TB_RECEBIMENTO_ITEM i "
                f"WHERE EXISTS ( "
                f"    SELECT 1 "
                f"    FROM TB_RECEBIMENTO r "
                f"    WHERE r.REC_EMPRESA = i.RIT_EMPRESA "
                f"      AND r.REC_ID = i.RIT_RECEBIMENTO "
                f"      AND r.REC_STATUS IN ('AB', 'PB') "
                f"      AND COALESCE(r.REC_ABERTO, 'S') = 'S' "
                f")"
            )

        campo_recebimento = self._obter_campo_vinculo_recebimento(tabela)

        if campo_recebimento:
            alias = "x"
            campo_empresa = None

            if tabela == "TB_BOLETO":
                campo_empresa = "BOL_EMPRESA"
            elif tabela == "TB_RECEBIMENTO_COBRANCA":
                campo_empresa = "RCC_EMPRESA"
            elif tabela == "TB_BORDERO_COMP_CARTAO_REC":
                campo_empresa = "BDCR_EMPRESA"
            elif tabela == "TB_BORDERO_COBRANCA_ITEM":
                campo_empresa = "BCI_EMPRESA"
            elif tabela == "TB_BORDERO_NEGOCIACAO_ITEM":
                campo_empresa = "BNI_EMPRESA"
            elif tabela == "TB_BORDERO_RECEBIMENTO_ITEM":
                campo_empresa = "BRI_EMPRESA"
            elif tabela == "TB_BORDERO_RESIDUO_REC":
                campo_empresa = "BDSR_EMPRESA"

            if campo_empresa:
                return (
                    f"SELECT {first}{alias}.* "
                    f"FROM {tabela} {alias} "
                    f"WHERE EXISTS ( "
                    f"    SELECT 1 "
                    f"    FROM TB_RECEBIMENTO r "
                    f"    WHERE r.REC_EMPRESA = {alias}.{campo_empresa} "
                    f"      AND r.REC_ID = {alias}.{campo_recebimento} "
                    f"      AND r.REC_STATUS IN ('AB', 'PB') "
                    f"      AND COALESCE(r.REC_ABERTO, 'S') = 'S' "
                    f")"
                )

        if self.var_somente_10.get():
            return f"SELECT FIRST 10 * FROM {tabela}"

        return f"SELECT * FROM {tabela}"

    def _contar_registros_origem_migracao(self, con_origem, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        if self.var_somente_10.get():
            return 10

        sql_count = self._montar_sql_count_origem_tabela(tabela)

        if not sql_count:
            return 10

        cur = con_origem.cursor()
        cur.execute(sql_count)
        linha = cur.fetchone()

        return int(linha[0] or 0)


    def _migrar_tabela(self, con_origem, con_destino, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        if tabela in self._obter_tabelas_financeiro_controlado():
            raise RuntimeError(
                f"A tabela {tabela} pertence ao financeiro em aberto e precisa da rotina "
                f"controlada com mapa REC_ID antigo -> novo. "
                f"Use apenas a análise por enquanto. A migração será liberada no próximo patch."
            )

        colunas_destino = [
            campo["nome"]
            for campo in self.firebird_service.obter_campos_tabela(con_destino, tabela)
        ]

        if not colunas_destino:
            raise RuntimeError(f"Nenhuma coluna encontrada na tabela destino {tabela}.")

        campo_chave_upsert = self._obter_chave_upsert_tabela(tabela)

        if self.var_limpar_destino.get() and not campo_chave_upsert:
            self._limpar_tabela_destino(con_destino, tabela)
            self._escrever_log(f"Destino limpo: {tabela}")

        cur_origem = con_origem.cursor()

        sql_origem = self._montar_sql_origem_tabela(tabela)

        if tabela in ("TB_RECEBIMENTO", "TB_RECEBIMENTO_ITEM"):
            self._escrever_log(f"[FILTRO] {tabela}: migrando somente financeiro em aberto.")

        cur_origem.execute(sql_origem)



        descricao = cur_origem.description
        if not descricao:
            return 0, 0

        colunas_origem = [desc[0].strip() for desc in descricao]

        registros = cur_origem.fetchall()

        total_registros_tabela = len(registros)

        if self.pb_execucao is not None:
            self.pb_execucao.configure(
                maximum=max(total_registros_tabela, 1),
                value=0,
            )

        lidos = 0
        inseridos = 0
        atualizados = 0

        for registro in registros:
            lidos += 1

            if self.lbl_progresso_execucao is not None:
                self.lbl_progresso_execucao.configure(
                    text=f"{tabela}: registro {lidos}/{total_registros_tabela}"
                )

            if self.pb_execucao is not None:
                self.pb_execucao.configure(value=lidos)

            if lidos == 1 or lidos % 50 == 0 or lidos == total_registros_tabela:
                self.update_idletasks()            

            dados = self._montar_dados_registro_por_colunas(
                registro,
                colunas_origem,
                colunas_destino,
            )

            if not dados:
                continue

            if campo_chave_upsert:
                campos_chave = campo_chave_upsert

                for campo_chave in campos_chave:
                    if campo_chave not in dados:
                        raise RuntimeError(
                            f"A chave {campo_chave} não foi encontrada nos dados da tabela {tabela}."
                        )

                existe = self._registro_existe_destino(
                    con_destino,
                    tabela,
                    campos_chave,
                    dados,
                )

                if existe:
                    self._executar_update_destino(
                        con_destino,
                        tabela,
                        dados,
                        campos_chave,
                    )
                    atualizados += 1
                else:
                    self._executar_insert_destino(

    def _testar_conexao(self, lado):
        con = None
        try:
            self.ambiente_firebird = self.firebird_service.detectar_ambiente_firebird()
            if self.lbl_status_conexao is not None:
                self.lbl_status_conexao.configure(text=self._montar_texto_ambiente())

            dados = self._obter_dados_conexao(lado)
            con, info = self._conectar_automatico(dados)

            cur = con.cursor()
            cur.execute("SELECT CURRENT_TIMESTAMP FROM RDB$DATABASE")
            linha = cur.fetchone()

            detalhe_cliente = "auto sem DLL fixa"
            if info["cliente"] and info["cliente"].get("dll"):
                detalhe_cliente = f"FB {info['cliente'].get('versao')} -> {info['cliente'].get('dll')}"

            messagebox.showinfo(
                "Conexão OK",
                f"Conexão {lado} realizada com sucesso.\n\n"
                f"Modo: {info['modo']}\n"
                f"Cliente: {detalhe_cliente}\n"
                f"Banco: {dados['banco']}\n"
                f"Data/hora do servidor: {linha[0]}",
            )
        except Exception as e:
            messagebox.showerror("Erro de conexão", f"Falha ao conectar na {lado}.\n\n{e}")
        finally:
            self._fechar_conexao(con)

    def _validar_conexoes(self):
        mensagens = []

        if not self.firebird_service.has_fdb():
            mensagens.append("- fdb não está instalado.")
        if not self.firebird_service.has_firebird_driver():
            mensagens.append("- firebird-driver não está instalado.")

        try:
            dados_origem = self._obter_dados_conexao("origem")
            if not dados_origem["banco"]:
                mensagens.append("- Informe o caminho do banco de origem.")
        except Exception as e:
            mensagens.append(f"- Origem: {e}")

        try:
            dados_destino = self._obter_dados_conexao("destino")
            if not dados_destino["banco"]:
                mensagens.append("- Informe o caminho do banco de destino.")
        except Exception as e:
            mensagens.append(f"- Destino: {e}")

        if mensagens:
            messagebox.showwarning(
                "Validação",
                "Ajuste os pontos abaixo antes de seguir:\n\n" + "\n".join(mensagens),
            )
            return

        con_origem = None
        con_destino = None

        try:
            con_origem, _ = self._conectar_automatico(dados_origem)
            con_destino, _ = self._conectar_automatico(dados_destino)

            self._escrever_log("Conexão origem validada com sucesso.")
            self._escrever_log("Conexão destino validada com sucesso.")
            self.lbl_resumo_execucao.configure(
                text="Conexões válidas. Já é possível analisar e iniciar a migração."
            )

            messagebox.showinfo(
                "Validação",
                "Origem e destino conectados com sucesso.",
            )
        except Exception as e:
            messagebox.showerror(
                "Erro na validação",
                f"Não foi possível validar as conexões.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con_origem)
            self._fechar_conexao(con_destino)

    def _registro_existe_destino(self, con_destino, tabela, campos_chave, dados):
        where = " AND ".join([f"{campo} = ?" for campo in campos_chave])
        valores = [dados[campo] for campo in campos_chave]

        cur = con_destino.cursor()
        cur.execute(
            f"SELECT FIRST 1 1 FROM {tabela} WHERE {where}",
            valores,
        )
        return cur.fetchone() is not None

    def _executar_update_destino(self, con_destino, tabela, dados, campos_chave):
        colunas_update = [col for col in dados.keys() if col not in campos_chave]

        if not colunas_update:
            return

        set_sql = ", ".join([f"{col} = ?" for col in colunas_update])
        where_sql = " AND ".join([f"{campo} = ?" for campo in campos_chave])

        sql = f"UPDATE {tabela} SET {set_sql} WHERE {where_sql}"

        valores = [dados[col] for col in colunas_update]
        valores.extend([dados[campo] for campo in campos_chave])

        cur = con_destino.cursor()
        cur.execute(sql, valores)
    def _limpar_valor_para_firebird(self, valor):
        if isinstance(valor, str):
            valor = valor.replace("\ufffd", "")
            valor = valor.replace("�", "")
            return valor

        return valor

    def _executar_insert_destino(self, con_destino, tabela, dados):
        if not dados:
            raise RuntimeError(f"Nenhum dado montado para inserir na tabela {tabela}.")

        colunas = list(dados.keys())
        placeholders = ", ".join(["?"] * len(colunas))
        sql = (
            f"INSERT INTO {tabela} ({', '.join(colunas)}) "
            f"VALUES ({placeholders})"
        )

        cur = con_destino.cursor()
        cur.execute(sql, [dados[col] for col in colunas])

    def _montar_dados_registro_por_colunas(self, registro, colunas_origem, colunas_destino):
        mapa_indices = {nome: i for i, nome in enumerate(colunas_origem)}
        colunas_comuns = [col for col in colunas_destino if col in mapa_indices]

        dados = {}
        for col in colunas_comuns:
            valor = registro[mapa_indices[col]]
            dados[col] = self._limpar_valor_para_firebird(valor)

        return dados

    def _montar_sql_origem_tabela(self, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        first = "FIRST 10 " if self.var_somente_10.get() else ""

        if tabela == "TB_RECEBIMENTO":
            return (
                f"SELECT {first}* "
                f"FROM TB_RECEBIMENTO "
                f"WHERE REC_STATUS IN ('AB', 'PB') "
                f"  AND COALESCE(REC_ABERTO, 'S') = 'S'"
            )

        if tabela == "TB_RECEBIMENTO_ITEM":
            return (
                f"SELECT {first}i.* "
                f"FROM TB_RECEBIMENTO_ITEM i "
                f"WHERE EXISTS ( "
                f"    SELECT 1 "
                f"    FROM TB_RECEBIMENTO r "
                f"    WHERE r.REC_EMPRESA = i.RIT_EMPRESA "
                f"      AND r.REC_ID = i.RIT_RECEBIMENTO "
                f"      AND r.REC_STATUS IN ('AB', 'PB') "
                f"      AND COALESCE(r.REC_ABERTO, 'S') = 'S' "
                f")"
            )

        if self.var_somente_10.get():
            return f"SELECT FIRST 10 * FROM {tabela}"

        return f"SELECT * FROM {tabela}"

    def _montar_sql_count_origem_tabela(self, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        if self.var_somente_10.get():
            return None

        if tabela == "TB_RECEBIMENTO":
            return (
                "SELECT COUNT(*) "
                "FROM TB_RECEBIMENTO "
                "WHERE REC_STATUS IN ('AB', 'PB') "
                "  AND COALESCE(REC_ABERTO, 'S') = 'S'"
            )

        if tabela == "TB_RECEBIMENTO_ITEM":
            return (
                "SELECT COUNT(*) "
                "FROM TB_RECEBIMENTO_ITEM i "
                "WHERE EXISTS ( "
                "    SELECT 1 "
                "    FROM TB_RECEBIMENTO r "
                "    WHERE r.REC_EMPRESA = i.RIT_EMPRESA "
                "      AND r.REC_ID = i.RIT_RECEBIMENTO "
                "      AND r.REC_STATUS IN ('AB', 'PB') "
                "      AND COALESCE(r.REC_ABERTO, 'S') = 'S' "
                ")"
            )

        return f"SELECT COUNT(*) FROM {tabela}"


    def _contar_registros_origem_migracao(self, con_origem, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        if self.var_somente_10.get():
            return 10

        sql_count = self._montar_sql_count_origem_tabela(tabela)

        if not sql_count:
            return 10

        cur = con_origem.cursor()
        cur.execute(sql_count)
        linha = cur.fetchone()

        return int(linha[0] or 0)


    def _migrar_tabela(self, con_origem, con_destino, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)

        colunas_destino = [
            campo["nome"]
            for campo in self.firebird_service.obter_campos_tabela(con_destino, tabela)
        ]

        if not colunas_destino:
            raise RuntimeError(f"Nenhuma coluna encontrada na tabela destino {tabela}.")

        campo_chave_upsert = self._obter_chave_upsert_tabela(tabela)

        if self.var_limpar_destino.get() and not campo_chave_upsert:
            self._limpar_tabela_destino(con_destino, tabela)
            self._escrever_log(f"Destino limpo: {tabela}")

        cur_origem = con_origem.cursor()

        sql_origem = self._montar_sql_origem_tabela(tabela)

        if tabela in ("TB_RECEBIMENTO", "TB_RECEBIMENTO_ITEM"):
            self._escrever_log(f"[FILTRO] {tabela}: migrando somente financeiro em aberto.")

        cur_origem.execute(sql_origem)



        descricao = cur_origem.description
        if not descricao:
            return 0, 0

        colunas_origem = [desc[0].strip() for desc in descricao]

        registros = cur_origem.fetchall()

        total_registros_tabela = len(registros)

        if self.pb_execucao is not None:
            self.pb_execucao.configure(
                maximum=max(total_registros_tabela, 1),
                value=0,
            )

        lidos = 0
        inseridos = 0
        atualizados = 0

        for registro in registros:
            lidos += 1

            if self.lbl_progresso_execucao is not None:
                self.lbl_progresso_execucao.configure(
                    text=f"{tabela}: registro {lidos}/{total_registros_tabela}"
                )

            if self.pb_execucao is not None:
                self.pb_execucao.configure(value=lidos)

            if lidos == 1 or lidos % 50 == 0 or lidos == total_registros_tabela:
                self.update_idletasks()            

            dados = self._montar_dados_registro_por_colunas(
                registro,
                colunas_origem,
                colunas_destino,
            )

            if not dados:
                continue

            if campo_chave_upsert:
                campos_chave = campo_chave_upsert

                for campo_chave in campos_chave:
                    if campo_chave not in dados:
                        raise RuntimeError(
                            f"A chave {campo_chave} não foi encontrada nos dados da tabela {tabela}."
                        )

                existe = self._registro_existe_destino(
                    con_destino,
                    tabela,
                    campos_chave,
                    dados,
                )

                if existe:
                    self._executar_update_destino(
                        con_destino,
                        tabela,
                        dados,
                        campos_chave,
                    )
                    atualizados += 1
                else:
                    self._executar_insert_destino(
                        con_destino,
                        tabela,
                        dados,
                    )
                    inseridos += 1

            else:
                self._executar_insert_destino(
                    con_destino,
                    tabela,
                    dados,
                )
                inseridos += 1

        if campo_chave_upsert:
            self._escrever_log(
                f"{tabela}: lidos={lidos} | inseridos={inseridos} | atualizados={atualizados}"
            )

        return lidos, inseridos

    def _limpar_tabela_destino(self, con_destino, tabela):
        tabela = self.firebird_service.normalizar_nome_tabela(tabela)
        cur = con_destino.cursor()
        cur.execute(f"DELETE FROM {tabela}")

    def _testar_conexao(self, lado):
        con = None
        try:
            self.ambiente_firebird = self.firebird_service.detectar_ambiente_firebird()
            if self.lbl_status_conexao is not None:
                self.lbl_status_conexao.configure(text=self._montar_texto_ambiente())

            dados = self._obter_dados_conexao(lado)
            con, info = self._conectar_automatico(dados)

            cur = con.cursor()
            cur.execute("SELECT CURRENT_TIMESTAMP FROM RDB$DATABASE")
            linha = cur.fetchone()

            detalhe_cliente = "auto sem DLL fixa"
            if info["cliente"] and info["cliente"].get("dll"):
                detalhe_cliente = f"FB {info['cliente'].get('versao')} -> {info['cliente'].get('dll')}"

            messagebox.showinfo(
                "Conexão OK",
                f"Conexão {lado} realizada com sucesso.\n\n"
                f"Modo: {info['modo']}\n"
                f"Cliente: {detalhe_cliente}\n"
                f"Banco: {dados['banco']}\n"
                f"Data/hora do servidor: {linha[0]}",
            )
        except Exception as e:
            messagebox.showerror("Erro de conexão", f"Falha ao conectar na {lado}.\n\n{e}")
        finally:
            self._fechar_conexao(con)

    def _ler_tabelas(self, lado):
        con = None
        try:
            dados = self._obter_dados_conexao(lado)
            con, _ = self._conectar_automatico(dados)

            tabelas = self.firebird_service.ler_tabelas(con)
            combo = self.cmb_origem_tabela if lado == "origem" else self.cmb_destino_tabela
            combo["values"] = tabelas

            if tabelas and not combo.get().strip():
                combo.set(tabelas[0])

            messagebox.showinfo(
                "Tabelas carregadas",
                f"Foram encontradas {len(tabelas)} tabelas na {lado}.",
            )
        except Exception as e:
            messagebox.showerror(
                "Erro ao ler tabelas",
                f"Não foi possível ler as tabelas da {lado}.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con)

    def _normalizar_nome_campo(self, nome):
        return re.sub(r"[^A-Z0-9]", "", (nome or "").upper())

    def _status_mapeamento(self, campo_origem, campo_destino):
        if not campo_destino:
            return "Sem destino"

        tipo_origem = campo_origem["tipo"]
        tipo_destino = campo_destino["tipo"]

        if tipo_origem == tipo_destino:
            return "Compatível"

        base_origem = tipo_origem.split("(")[0]
        base_destino = tipo_destino.split("(")[0]

        if base_origem == base_destino:
            return "Compatível parcial"

        return "Tipo diferente"

    def _montar_mapeamento_automatico(self, campos_origem, campos_destino):
        indice_destino = {
            self._normalizar_nome_campo(campo["nome"]): campo
            for campo in campos_destino
        }

        linhas = []
        for campo_origem in campos_origem:
            chave = self._normalizar_nome_campo(campo_origem["nome"])
            campo_destino = indice_destino.get(chave)

            linhas.append(
                {
                    "origem": campo_origem["nome"],
                    "tipo_origem": campo_origem["tipo"],
                    "destino": campo_destino["nome"] if campo_destino else "",
                    "tipo_destino": campo_destino["tipo"] if campo_destino else "",
                    "regra": "Copiar" if campo_destino else "Ignorar",
                    "valor_fixo": "",
                    "status": self._status_mapeamento(campo_origem, campo_destino),
                }
            )

        return linhas

        }

    def _extrair_termo_sugerido(self, nome_campo):
        texto = (nome_campo or "").upper().strip()

        partes = [p for p in re.split(r"[^A-Z0-9]+", texto) if p]

        ignorar = {
            "CLI",
            "TB",
            "ID",
            "COD",
            "CODIGO",
            "DT",
            "DATA",
            "NUM",
            "NUMERO",
            "NR",
        }

        partes_validas = [p for p in partes if p not in ignorar]

        if not partes_validas:
            partes_validas = partes

        if not partes_validas:
            return ""

        principal = max(partes_validas, key=len)

        if len(principal) >= 5:
            return principal[:4]

        return principal

            if not base.exists():
                continue

            for pasta in base.rglob("*"):
                if not pasta.is_dir():
                    continue

                nome = pasta.name.upper()
                if "FIREBIRD" not in nome:
                    continue

                dll = pasta / "fbclient.dll"
                if not dll.exists():
                    dll = pasta / "bin" / "fbclient.dll"
                    if not dll.exists():
                        continue

                versao = self._extrair_versao_firebird(str(pasta))
                conf = dll.parent / "firebird.conf"
                if not conf.exists():
                    conf = dll.parent.parent / "firebird.conf"

                porta = self._ler_porta_firebird_conf(conf)
                item = {
                    "versao": versao,
                    "dll": str(dll),
                    "porta": porta,
                    "pasta": str(pasta),
                    "familia": self._familia_firebird(versao),
                }

                chave = (item["versao"], item["dll"])
                if chave not in vistos:
                    vistos.add(chave)
                    candidatos.append(item)

        candidatos.sort(key=lambda x: (self._versao_para_numero(x["versao"]), x["dll"]))
        return candidatos

    def montar_texto_ambiente(self, ambiente_firebird):
        itens = []

        if not ambiente_firebird:
            itens.append("Nenhuma instalação local do Firebird foi detectada automaticamente.")
        else:
            partes = []
            for item in ambiente_firebird:
                porta = item["porta"] or "?"
                partes.append(f"FB {item['versao']} (porta {porta})")
            itens.append("Ambiente Firebird detectado: " + ", ".join(partes) + ".")

        itens.append("fdb OK." if fdb is not None else "fdb não instalado.")
        itens.append("firebird-driver OK." if fb_driver_connect is not None else "firebird-driver não instalado.")
        return "  ".join(itens)

    def porta_padrao_detectada(self, ambiente_firebird):
        for item in reversed(ambiente_firebird):
            if item["porta"]:
                return item["porta"]
        return "3050"

    def conectar_automatico(self, dados, ambiente_firebird):
        if not dados["banco"]:
            raise ValueError("Informe o caminho do banco.")

        cliente = self._resolver_cliente_preferencial(dados["porta"], ambiente_firebird)
        erros = []
        tentativas = []

        if cliente:
            if cliente["familia"] == "legacy":
                tentativas.append(("fdb", cliente))
                tentativas.append(("driver", None))
            else:
                tentativas.append(("driver", cliente))
                tentativas.append(("fdb", cliente))
        else:
            tentativas.append(("driver", None))
            tentativas.append(("fdb", None))

        usados = set()
        tentativas_filtradas = []

        for modo, cli in tentativas:
            chave = (modo, cli["dll"] if cli else None)
            if chave not in usados:
                usados.add(chave)
                tentativas_filtradas.append((modo, cli))

        for modo, cli in tentativas_filtradas:
            try:
                if modo == "driver":
                    con = self._conectar_firebird_driver(dados, cli)
                    return con, {"modo": "firebird-driver", "cliente": cli}

                if modo == "fdb":
                    con = self._conectar_fdb(dados, cli)
                    return con, {"modo": "fdb", "cliente": cli}
            except Exception as e:
                detalhe = modo
                if cli and cli.get("versao"):
                    detalhe += f" / FB {cli['versao']}"
                erros.append(f"{detalhe}: {e}")

        raise RuntimeError(
            "Nenhuma estratégia automática de conexão funcionou.\n\n"
            + "\n\n".join(erros)
        )

    def ler_tabelas(self, con):
        cur = con.cursor()
        cur.execute(
            """
            SELECT TRIM(RDB$RELATION_NAME)
            FROM RDB$RELATIONS
            WHERE COALESCE(RDB$SYSTEM_FLAG, 0) = 0
              AND RDB$VIEW_BLR IS NULL
            ORDER BY RDB$RELATION_NAME
            """
        )
        return [linha[0].strip() for linha in cur.fetchall() if linha[0]]

    def obter_campos_tabela(self, con, nome_tabela):
        nome_tabela = self.normalizar_nome_tabela(nome_tabela)
        cur = con.cursor()

        cur.execute(
            """
            SELECT
                TRIM(rf.RDB$FIELD_NAME) AS CAMPO,
                f.RDB$FIELD_TYPE,
                f.RDB$FIELD_SUB_TYPE,
                f.RDB$FIELD_LENGTH,
                f.RDB$FIELD_PRECISION,
                f.RDB$FIELD_SCALE,
                rf.RDB$NULL_FLAG
            FROM RDB$RELATION_FIELDS rf
            JOIN RDB$FIELDS f
              ON f.RDB$FIELD_NAME = rf.RDB$FIELD_SOURCE
            WHERE TRIM(rf.RDB$RELATION_NAME) = ?
            ORDER BY rf.RDB$FIELD_POSITION
            """,
            (nome_tabela,),
        )

        campos = []
        for linha in cur.fetchall():
            campo = (linha[0] or "").strip()
            field_type = linha[1]
            field_length = linha[3]
            field_precision = linha[4]
            field_scale = linha[5]
            null_flag = linha[6]

            tipo_texto = self._traduzir_tipo_firebird(
                field_type,
                field_length,
                field_precision,
                field_scale,
            )

            campos.append(
                {
                    "nome": campo,
                    "tipo": tipo_texto,
                    "obrigatorio": str(null_flag or "").strip() == "1",
                }
            )

        return campos

    def normalizar_nome_tabela(self, nome_tabela):
        nome_tabela = (nome_tabela or "").strip().upper()

        if not nome_tabela:
            raise ValueError("Informe uma tabela.")

        if not re.fullmatch(r"[A-Z0-9_$]+", nome_tabela):
            raise ValueError(
                "Nome de tabela inválido. Use uma tabela carregada pelo botão 'Ler tabelas'."
            )

        return nome_tabela

    def _traduzir_tipo_firebird(self, field_type, field_length, field_precision, field_scale):
        mapa = {
            7: "SMALLINT",
            8: "INTEGER",
            10: "FLOAT",
            12: "DATE",
            13: "TIME",
            14: "CHAR",
            16: "BIGINT/NUMERIC",
            27: "DOUBLE",
            35: "TIMESTAMP",
            37: "VARCHAR",
            261: "BLOB",
        }

        base = mapa.get(field_type, f"TIPO_{field_type}")

        if field_type in (14, 37):
            return f"{base}({field_length})"

        if field_type == 16:
            if field_scale and int(field_scale) < 0:
                escala = abs(int(field_scale))
                if field_precision:
                    return f"NUMERIC({field_precision},{escala})"
                return f"NUMERIC(*,{escala})"
            return "BIGINT"

        return base

    def _extrair_versao_firebird(self, texto):
        texto_upper = (texto or "").upper().replace("-", "_")

        padroes = [
            r"FIREBIRD[_\s]?(\d+)[._](\d+)",
            r"FIREBIRD[_\s]?(\d+)",
        ]

        for padrao in padroes:
            m = re.search(padrao, texto_upper)
            if m:
                if len(m.groups()) >= 2:
                    return f"{m.group(1)}.{m.group(2)}"
                return m.group(1)

        return "desconhecida"

    def _familia_firebird(self, versao):
        try:
            major = int(float(str(versao)))
        except Exception:
            return "desconhecida"

        if major <= 2:
            return "legacy"
        return "moderno"

    def _versao_para_numero(self, versao):
        try:
            return float(str(versao))
        except Exception:
            return 0.0

    def _ler_porta_firebird_conf(self, caminho_conf):
        try:
            if not caminho_conf or not Path(caminho_conf).exists():
                return None

            linhas = Path(caminho_conf).read_text(
                encoding="utf-8",
                errors="ignore",
            ).splitlines()

            for linha in linhas:
                linha_limpa = linha.strip()
                if not linha_limpa or linha_limpa.startswith("#"):
                    continue
                if linha_limpa.upper().startswith("REMOTESERVICEPORT"):
                    partes = linha_limpa.split("=")
                    if len(partes) == 2:
                        return partes[1].strip()
        except Exception:
            pass

        return None

    def _resolver_cliente_preferencial(self, porta, ambiente_firebird):
        porta_txt = str(porta).strip()

        if ambiente_firebird:
            ordenados = sorted(
                ambiente_firebird,
                key=lambda x: self._versao_para_numero(x["versao"]),
                reverse=True,
            )

            for item in ordenados:
                if item["porta"] and str(item["porta"]) == porta_txt:
                    return item

            for item in ordenados:
                if porta_txt == "3050" and item["familia"] == "legacy":
                    return item

            for item in ordenados:
                if item["familia"] == "moderno":
                    return item

            return ambiente_firebird[-1]

        return None

    def _conectar_fdb(self, dados, cliente=None):
        if fdb is None:
            raise RuntimeError("Biblioteca Python 'fdb' não está instalada. Rode: pip install fdb")

        kwargs = {
            "dsn": f"{dados['servidor']}/{dados['porta']}:{dados['banco']}",
            "user": dados["usuario"],
            "password": dados["senha"],
            "charset": dados["charset"],
        }

        if cliente and cliente.get("dll"):
            kwargs["fb_library_name"] = cliente["dll"]

        return fdb.connect(**kwargs)

    def _conectar_firebird_driver(self, dados, cliente=None):
        if fb_driver_connect is None:
            raise RuntimeError(
                "Biblioteca Python 'firebird-driver' não está instalada. Rode: pip install firebird-driver"
            )

        dsn = f"{dados['servidor']}/{dados['porta']}:{dados['banco']}"

        if driver_config is not None:
            try:
                if cliente and cliente.get("dll"):
                    driver_config.fb_client_library.value = cliente["dll"]
                else:
                    driver_config.fb_client_library.value = None
            except Exception:
                pass

        kwargs = {
            "user": dados["usuario"],
            "password": dados["senha"],
            "charset": dados["charset"],
        }

        try:
            return fb_driver_connect(
                dsn=dsn,
                **kwargs,
            )
        except TypeError:
            return fb_driver_connect(
                database=dsn,
                **kwargs,
            )

    def _detectar_ambiente_firebird(self):
        candidatos = []
        vistos = set()

        bases = [
            Path(r"C:\Program Files\Firebird"),
            Path(r"C:\Program Files (x86)\Firebird"),
            Path(r"C:\Firebird"),
        ]

        for base in bases:
            if not base.exists():
                continue

            for pasta in base.rglob("*"):
                if not pasta.is_dir():
                    continue

                nome = pasta.name.upper()
                if "FIREBIRD" not in nome:
                    continue

                dll = pasta / "fbclient.dll"
                if not dll.exists():
                    dll = pasta / "bin" / "fbclient.dll"
                    if not dll.exists():
                        continue

                versao = self._extrair_versao_firebird(str(pasta))
                conf = dll.parent / "firebird.conf"
                if not conf.exists():
                    conf = dll.parent.parent / "firebird.conf"

                porta = self._ler_porta_firebird_conf(conf)
                item = {
                    "versao": versao,
                    "dll": str(dll),
                    "porta": porta,
                    "pasta": str(pasta),
                    "familia": self._familia_firebird(versao),
                }

                chave = (item["versao"], item["dll"])
                if chave not in vistos:
                    vistos.add(chave)
                    candidatos.append(item)

        candidatos.sort(key=lambda x: (self._versao_para_numero(x["versao"]), x["dll"]))
        return candidatos

    def _extrair_versao_firebird(self, texto):
        texto_upper = (texto or "").upper().replace("-", "_")

        padroes = [
            r"FIREBIRD[_\s]?(\d+)[._](\d+)",
            r"FIREBIRD[_\s]?(\d+)",
        ]

        for padrao in padroes:
            m = re.search(padrao, texto_upper)
            if m:
                if len(m.groups()) >= 2:
                    return f"{m.group(1)}.{m.group(2)}"
                return m.group(1)

        return "desconhecida"

    def _familia_firebird(self, versao):
        try:
            major = int(float(str(versao)))
        except Exception:
            return "desconhecida"

        if major <= 2:
            return "legacy"
        return "moderno"

    def _versao_para_numero(self, versao):
        try:
            return float(str(versao))
        except Exception:
            return 0.0

    def _ler_porta_firebird_conf(self, caminho_conf):
        try:
            if not caminho_conf or not Path(caminho_conf).exists():
                return None

            linhas = Path(caminho_conf).read_text(
                encoding="utf-8",
                errors="ignore",
            ).splitlines()

            for linha in linhas:
                linha_limpa = linha.strip()
                if not linha_limpa or linha_limpa.startswith("#"):
                    continue
                if linha_limpa.upper().startswith("REMOTESERVICEPORT"):
                    partes = linha_limpa.split("=")
                    if len(partes) == 2:
                        return partes[1].strip()
        except Exception:
            pass

        return None

    def _montar_texto_ambiente(self):
        itens = []

        if not self.ambiente_firebird:
            itens.append("Nenhuma instalação local do Firebird foi detectada automaticamente.")
        else:
            partes = []
            for item in self.ambiente_firebird:
                porta = item["porta"] or "?"
                partes.append(f"FB {item['versao']} (porta {porta})")
            itens.append("Ambiente Firebird detectado: " + ", ".join(partes) + ".")

        itens.append("fdb OK." if fdb is not None else "fdb não instalado.")
        itens.append("firebird-driver OK." if fb_driver_connect is not None else "firebird-driver não instalado.")
        return "  ".join(itens)

    def _porta_padrao_detectada(self):
        for item in reversed(self.ambiente_firebird):
            if item["porta"]:
                return item["porta"]
        return "3050"

    def _resolver_cliente_preferencial(self, porta):
        porta_txt = str(porta).strip()

        if self.ambiente_firebird:
            ordenados = sorted(
                self.ambiente_firebird,
                key=lambda x: self._versao_para_numero(x["versao"]),
                reverse=True,
            )

            for item in ordenados:
                if item["porta"] and str(item["porta"]) == porta_txt:
                    return item

            for item in ordenados:
                if porta_txt == "3050" and item["familia"] == "legacy":
                    return item

            for item in ordenados:
                if item["familia"] == "moderno":
                    return item

            return self.ambiente_firebird[-1]

        return None

    def _conectar_automatico(self, dados):
        if not dados["banco"]:
            raise ValueError("Informe o caminho do banco.")

        cliente = self._resolver_cliente_preferencial(dados["porta"])
        erros = []
        tentativas = []

        if cliente:
            if cliente["familia"] == "legacy":
                tentativas.append(("fdb", cliente))
                tentativas.append(("driver", None))
            else:
                tentativas.append(("driver", cliente))
                tentativas.append(("fdb", None))
        else:
            tentativas.append(("driver", None))
            tentativas.append(("fdb", None))

        usados = set()
        tentativas_filtradas = []

        for modo, cli in tentativas:
            chave = (modo, cli["dll"] if cli else None)
            if chave not in usados:
                usados.add(chave)
                tentativas_filtradas.append((modo, cli))

        for modo, cli in tentativas_filtradas:
            try:
                if modo == "driver":
                    con = self._conectar_firebird_driver(dados, cli)
                    return con, {"modo": "firebird-driver", "cliente": cli}

                if modo == "fdb":
                    con = self._conectar_fdb(dados, cli)
                    return con, {"modo": "fdb", "cliente": cli}
            except Exception as e:
                detalhe = modo
                if cli and cli.get("versao"):
                    detalhe += f" / FB {cli['versao']}"
                erros.append(f"{detalhe}: {e}")

        raise RuntimeError(
            "Nenhuma estratégia automática de conexão funcionou.\n\n"
            + "\n\n".join(erros)
        )

    def _conectar_fdb(self, dados, cliente=None):
        if fdb is None:
            raise RuntimeError("Biblioteca Python 'fdb' não está instalada. Rode: pip install fdb")

        kwargs = {
            "dsn": f"{dados['servidor']}/{dados['porta']}:{dados['banco']}",
            "user": dados["usuario"],
            "password": dados["senha"],
            "charset": dados["charset"],
        }

        if cliente and cliente.get("dll"):
            kwargs["fb_library_name"] = cliente["dll"]

        return fdb.connect(**kwargs)

    def _conectar_firebird_driver(self, dados, cliente=None):
        if fb_driver_connect is None:
            raise RuntimeError(
                "Biblioteca Python 'firebird-driver' não está instalada. Rode: pip install firebird-driver"
            )

        if driver_config is not None:
            try:
                driver_config.fb_client_library.value = (
                    cliente["dll"] if cliente and cliente.get("dll") else None
                )
            except Exception:
                pass

        dsn = f"{dados['servidor']}/{dados['porta']}:{dados['banco']}"

        try:
            return fb_driver_connect(
                dsn=dsn,
                user=dados["usuario"],
                password=dados["senha"],
                charset=dados["charset"],
            )
        except TypeError:
            return fb_driver_connect(
                database=dsn,
                user=dados["usuario"],
                password=dados["senha"],
                charset=dados["charset"],
            )

    def _testar_conexao(self, lado):
        con = None
        try:
            dados = self._obter_dados_conexao(lado)
            con, info = self._conectar_automatico(dados)

            cur = con.cursor()
            cur.execute("SELECT CURRENT_TIMESTAMP FROM RDB$DATABASE")
            linha = cur.fetchone()

            detalhe_cliente = "auto sem DLL fixa"
            if info["cliente"] and info["cliente"].get("dll"):
                detalhe_cliente = f"FB {info['cliente'].get('versao')} -> {info['cliente'].get('dll')}"

            messagebox.showinfo(
                "Conexão OK",
                f"Conexão {lado} realizada com sucesso.\n\n"
                f"Modo: {info['modo']}\n"
                f"Cliente: {detalhe_cliente}\n"
                f"Banco: {dados['banco']}\n"
                f"Data/hora do servidor: {linha[0]}",
            )
        except Exception as e:
            messagebox.showerror("Erro de conexão", f"Falha ao conectar na {lado}.\n\n{e}")
        finally:
            self._fechar_conexao(con)

    def _ler_tabelas(self, lado):
        con = None
        try:
            dados = self._obter_dados_conexao(lado)
            con, _ = self._conectar_automatico(dados)
            cur = con.cursor()

            cur.execute(
                """
                SELECT TRIM(RDB$RELATION_NAME)
                FROM RDB$RELATIONS
                WHERE COALESCE(RDB$SYSTEM_FLAG, 0) = 0
                  AND RDB$VIEW_BLR IS NULL
                ORDER BY RDB$RELATION_NAME
                """
            )

            tabelas = [linha[0].strip() for linha in cur.fetchall() if linha[0]]
            combo = self.cmb_origem_tabela if lado == "origem" else self.cmb_destino_tabela
            combo["values"] = tabelas

            if tabelas and not combo.get().strip():
                combo.set(tabelas[0])

            messagebox.showinfo(
                "Tabelas carregadas",
                f"Foram encontradas {len(tabelas)} tabelas na {lado}.",
            )
        except Exception as e:
            messagebox.showerror(
                "Erro ao ler tabelas",
                f"Não foi possível ler as tabelas da {lado}.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con)

    def _obter_campos_tabela(self, con, nome_tabela):
        nome_tabela = self._normalizar_nome_tabela(nome_tabela)
        cur = con.cursor()

        cur.execute(
            """
            SELECT
                TRIM(rf.RDB$FIELD_NAME) AS CAMPO,
                f.RDB$FIELD_TYPE,
                f.RDB$FIELD_SUB_TYPE,
                f.RDB$FIELD_LENGTH,
                f.RDB$FIELD_PRECISION,
                f.RDB$FIELD_SCALE,
                rf.RDB$NULL_FLAG
            FROM RDB$RELATION_FIELDS rf
            JOIN RDB$FIELDS f
              ON f.RDB$FIELD_NAME = rf.RDB$FIELD_SOURCE
            WHERE TRIM(rf.RDB$RELATION_NAME) = ?
            ORDER BY rf.RDB$FIELD_POSITION
            """,
            (nome_tabela,),
        )

        campos = []
        for linha in cur.fetchall():
            campo = (linha[0] or "").strip()
            field_type = linha[1]
            field_length = linha[3]
            field_precision = linha[4]
            field_scale = linha[5]
            null_flag = linha[6]

            tipo_texto = self._traduzir_tipo_firebird(
                field_type,
                field_length,
                field_precision,
                field_scale,
            )

            campos.append(
                {
                    "nome": campo,
                    "tipo": tipo_texto,
                    "obrigatorio": str(null_flag or "").strip() == "1",
                }
            )

        return campos

    def _traduzir_tipo_firebird(self, field_type, field_length, field_precision, field_scale):
        mapa = {
            7: "SMALLINT",
            8: "INTEGER",
            10: "FLOAT",
            12: "DATE",
            13: "TIME",
            14: "CHAR",
            16: "BIGINT/NUMERIC",
            27: "DOUBLE",
            35: "TIMESTAMP",
            37: "VARCHAR",
            261: "BLOB",
        }

        base = mapa.get(field_type, f"TIPO_{field_type}")

        if field_type in (14, 37):
            return f"{base}({field_length})"

        if field_type == 16:
            if field_scale and int(field_scale) < 0:
                escala = abs(int(field_scale))
                if field_precision:
                    return f"NUMERIC({field_precision},{escala})"
                return f"NUMERIC(*,{escala})"
            return "BIGINT"

        return base

    def _normalizar_nome_campo(self, nome):
        return re.sub(r"[^A-Z0-9]", "", (nome or "").upper())

    def _status_mapeamento(self, campo_origem, campo_destino):
        if not campo_destino:
            return "Sem destino"

        tipo_origem = campo_origem["tipo"]
        tipo_destino = campo_destino["tipo"]

        if tipo_origem == tipo_destino:
            return "Compatível"

        base_origem = tipo_origem.split("(")[0]
        base_destino = tipo_destino.split("(")[0]

        if base_origem == base_destino:
            return "Compatível parcial"

        return "Tipo diferente"

    def _montar_mapeamento_automatico(self, campos_origem, campos_destino):
        indice_destino = {
            self._normalizar_nome_campo(campo["nome"]): campo
            for campo in campos_destino
        }

        linhas = []
        for campo_origem in campos_origem:
            chave = self._normalizar_nome_campo(campo_origem["nome"])
            campo_destino = indice_destino.get(chave)

            linhas.append(
                {
                    "origem": campo_origem["nome"],
                    "tipo_origem": campo_origem["tipo"],
                    "destino": campo_destino["nome"] if campo_destino else "",
                    "tipo_destino": campo_destino["tipo"] if campo_destino else "",
                    "status": self._status_mapeamento(campo_origem, campo_destino),
                }
            )

        return linhas
    
    def _obter_nomes_destino_usados(self):
        return {
            (linha.get("destino") or "").strip()
            for linha in self.mapeamento_linhas
            if (linha.get("destino") or "").strip()
        }  

    def _obter_texto_pesquisa_campos(self, lado):
        if lado == "origem":
            if self.edt_pesquisa_origem is None:
                return ""
            return (self.edt_pesquisa_origem.get() or "").strip().upper()

        if self.edt_pesquisa_destino is None:
            return ""
        return (self.edt_pesquisa_destino.get() or "").strip().upper()

    def _filtrar_campos(self, lado):
        if lado == "origem":
            self._preencher_tree_campos("origem", self.campos_origem)
        else:
            self._preencher_tree_campos("destino", self.campos_destino)

    def _limpar_pesquisa_campos(self, lado):
        if lado == "origem":
            if self.edt_pesquisa_origem is not None:
                self.edt_pesquisa_origem.delete(0, tk.END)
            self._preencher_tree_campos("origem", self.campos_origem)
        else:
            if self.edt_pesquisa_destino is not None:
                self.edt_pesquisa_destino.delete(0, tk.END)
            self._preencher_tree_campos("destino", self.campos_destino)    


    def _preencher_tree_campos(self, lado, campos):
        tree = self.tree_campos_origem if lado == "origem" else self.tree_campos_destino

        for item in tree.get_children():
            tree.delete(item)

        destinos_usados = self._obter_nomes_destino_usados()
        filtro = self._obter_texto_pesquisa_campos(lado)

        indice_visual = 0

        for campo in campos:
            nome_campo = (campo["nome"] or "").upper()
            tipo_campo = (campo["tipo"] or "").upper()

            if filtro:
                if filtro not in nome_campo and filtro not in tipo_campo:
                    continue
            tags = ()

            if lado == "origem":
                linha_mapeada = next(
                    (
                        linha
                        for linha in self.mapeamento_linhas
                        if (linha.get("origem") or "").strip() == (campo["nome"] or "").strip()
                    ),
                    None,
                )

                if linha_mapeada and (linha_mapeada.get("status") or "").strip() == "Sem destino":
                    tags = ("pendente",)
            else:
                if (campo["nome"] or "").strip() not in destinos_usados:
                    tags = ("pendente",)

            tree.insert(
                "",
                "end",
                iid=str(indice_visual),
                values=(campo["nome"], campo["tipo"]),
                tags=tags,
            )

            indice_visual += 1

        tree.tag_configure("pendente", background="#ffd6d6")

    def _preencher_tree_mapeamento(self):
        for item in self.tree_mapeamento.get_children():
            self.tree_mapeamento.delete(item)

        for i, linha in enumerate(self.mapeamento_linhas):
            self.tree_mapeamento.insert(
                "",
                "end",
                iid=str(i),
                values=(
                    linha["origem"],
                    linha["tipo_origem"],
                    linha["destino"],
                    linha["tipo_destino"],
                    linha["status"],
                ),
            )

    def _atualizar_status_resumo_mapeamento(self):
        compat = sum(1 for x in self.mapeamento_linhas if x["status"] == "Compatível")
        parcial = sum(1 for x in self.mapeamento_linhas if x["status"] == "Compatível parcial")
        dif = sum(1 for x in self.mapeamento_linhas if x["status"] == "Tipo diferente")
        sem_destino = sum(1 for x in self.mapeamento_linhas if x["status"] == "Sem destino")

        self.lbl_status_mapeamento.configure(
            text=(
                f"Mapeamento gerado: {len(self.mapeamento_linhas)} campo(s) origem | "
                f"Compatíveis: {compat} | "
                f"Compatíveis parciais: {parcial} | "
                f"Tipo diferente: {dif} | "
                f"Sem destino: {sem_destino}"
            )
        )

        if self.campos_origem:
            self._preencher_tree_campos("origem", self.campos_origem)

        if self.campos_destino:
            self._preencher_tree_campos("destino", self.campos_destino)        

    def _ao_selecionar_campo_origem(self, event=None):
        selecionados = self.tree_campos_origem.selection()
        if not selecionados:
            self.campo_origem_selecionado = None
            self.lbl_origem_selecionado.configure(text="(nenhum)")
            return

        valores = self.tree_campos_origem.item(selecionados[0], "values")
        nome_campo = valores[0] if valores else ""

        self.campo_origem_selecionado = self._buscar_campo_origem_por_nome(nome_campo)
        self.lbl_origem_selecionado.configure(
            text=self.campo_origem_selecionado["nome"] if self.campo_origem_selecionado else "(nenhum)"
        )

    def _ao_selecionar_campo_destino(self, event=None):
        selecionados = self.tree_campos_destino.selection()
        if not selecionados:
            self.campo_destino_selecionado = None
            self.lbl_destino_selecionado.configure(text="(nenhum)")
            return

        valores = self.tree_campos_destino.item(selecionados[0], "values")
        nome_campo = valores[0] if valores else ""

        self.campo_destino_selecionado = self._buscar_campo_destino_por_nome(nome_campo)
        self.lbl_destino_selecionado.configure(
            text=self.campo_destino_selecionado["nome"] if self.campo_destino_selecionado else "(nenhum)"
        )

    def _ao_selecionar_linha_mapeamento(self, event=None):
        selecionados = self.tree_mapeamento.selection()
        if not selecionados:
            self.indice_mapeamento_selecionado = None
            return

        self.indice_mapeamento_selecionado = int(selecionados[0])

    def _buscar_campo_origem_por_nome(self, nome):
        for campo in self.campos_origem:
            if campo["nome"] == nome:
                return campo
        return None

    def _buscar_campo_destino_por_nome(self, nome):
        for campo in self.campos_destino:
            if campo["nome"] == nome:
                return campo
        return None

    def _vincular_campos_selecionados(self):
        if not self.campo_origem_selecionado:
            messagebox.showwarning("Mapeamento", "Selecione um campo da origem.")
            return

        if not self.campo_destino_selecionado:
            messagebox.showwarning("Mapeamento", "Selecione um campo do destino.")
            return

        indice = None
        for i, linha in enumerate(self.mapeamento_linhas):
            if linha["origem"] == self.campo_origem_selecionado["nome"]:
                indice = i
                break

        if indice is None:
            messagebox.showwarning(
                "Mapeamento",
                "Não foi encontrada linha de mapeamento para o campo origem selecionado.",
            )
            return

        self.mapeamento_linhas[indice]["destino"] = self.campo_destino_selecionado["nome"]
        self.mapeamento_linhas[indice]["tipo_destino"] = self.campo_destino_selecionado["tipo"]
        self.mapeamento_linhas[indice]["status"] = self._status_mapeamento(
            self.campo_origem_selecionado,
            self.campo_destino_selecionado,
        )

        self._preencher_tree_mapeamento()
        self._atualizar_status_resumo_mapeamento()

        self.tree_mapeamento.selection_set(str(indice))
        self.tree_mapeamento.focus(str(indice))
        self.indice_mapeamento_selecionado = indice

        self.lbl_status_vinculo.configure(
            text=f"Vinculado: {self.campo_origem_selecionado['nome']} -> {self.campo_destino_selecionado['nome']}"
        )

    def _limpar_mapeamento_manual(self):
        if self.indice_mapeamento_selecionado is None:
            messagebox.showwarning("Mapeamento", "Selecione uma linha no grid de mapeamento.")
            return

        linha = self.mapeamento_linhas[self.indice_mapeamento_selecionado]
        linha["destino"] = ""
        linha["tipo_destino"] = ""
        linha["status"] = "Sem destino"

        self._preencher_tree_mapeamento()
        self._atualizar_status_resumo_mapeamento()

        self.tree_mapeamento.selection_set(str(self.indice_mapeamento_selecionado))
        self.tree_mapeamento.focus(str(self.indice_mapeamento_selecionado))

        self.lbl_status_vinculo.configure(text="Vínculo removido da linha selecionada.")

    def _remontar_mapeamento_automatico(self):
        if not self.campos_origem or not self.campos_destino:
            messagebox.showwarning(
                "Mapeamento",
                "Primeiro valide origem e destino para carregar os campos.",
            )
            return

        self.mapeamento_linhas = self._montar_mapeamento_automatico(
            self.campos_origem,
            self.campos_destino,
        )
        self._preencher_tree_mapeamento()
        self._atualizar_status_resumo_mapeamento()
        self.lbl_status_vinculo.configure(text="Mapeamento automático remontado.")

    def _validar_ambiente(self):
        mensagens = []

        if fdb is None:
            mensagens.append("- fdb não está instalado.")
        if fb_driver_connect is None:
            mensagens.append("- firebird-driver não está instalado.")

        try:
            dados_origem = self._obter_dados_conexao("origem")
            if not dados_origem["banco"]:
                mensagens.append("- Informe o caminho do banco de origem.")
            if not dados_origem["tabela"]:
                mensagens.append("- Escolha uma tabela de origem.")
        except Exception as e:
            mensagens.append(f"- Origem: {e}")

        try:
            dados_destino = self._obter_dados_conexao("destino")
            if not dados_destino["banco"]:
                mensagens.append("- Informe o caminho do banco de destino.")
            if not dados_destino["tabela"]:
                mensagens.append("- Escolha uma tabela de destino.")
        except Exception as e:
            mensagens.append(f"- Destino: {e}")

        if mensagens:
            messagebox.showwarning(
                "Validação",
                "Ajuste os pontos abaixo antes de seguir:\n\n" + "\n".join(mensagens),
            )
            return

        con_origem = None
        con_destino = None

        try:
            con_origem, _ = self._conectar_automatico(dados_origem)
            con_destino, _ = self._conectar_automatico(dados_destino)

            self.campos_origem = self._obter_campos_tabela(con_origem, dados_origem["tabela"])
            self.campos_destino = self._obter_campos_tabela(con_destino, dados_destino["tabela"])

            self._preencher_tree_campos("origem", self.campos_origem)
            self._preencher_tree_campos("destino", self.campos_destino)

            self.mapeamento_linhas = self._montar_mapeamento_automatico(
                self.campos_origem,
                self.campos_destino,
            )
            self._preencher_tree_mapeamento()
            self._atualizar_status_resumo_mapeamento()

            self.campo_origem_selecionado = None
            self.campo_destino_selecionado = None
            self.indice_mapeamento_selecionado = None
            self.lbl_origem_selecionado.configure(text="(nenhum)")
            self.lbl_destino_selecionado.configure(text="(nenhum)")
            self.lbl_status_vinculo.configure(text="Campos carregados. Agora compare os dois lados e faça os vínculos manuais quando precisar.")

            messagebox.showinfo(
                "Validação",
                "Conexões e tabelas válidas.\n\nOs campos da origem e do destino foram carregados na aba de mapeamento.",
            )
        except Exception as e:
            messagebox.showerror(
                "Erro na validação",
                f"Não foi possível validar e montar o mapeamento.\n\n{e}",
            )
        finally:
            self._fechar_conexao(con_origem)
            self._fechar_conexao(con_destino)

    def _normalizar_nome_tabela(self, nome_tabela):
        nome_tabela = (nome_tabela or "").strip().upper()

        if not nome_tabela:
            raise ValueError("Informe uma tabela.")

        if not re.fullmatch(r"[A-Z0-9_$]+", nome_tabela):
            raise ValueError(
                "Nome de tabela inválido. Use uma tabela carregada pelo botão 'Ler tabelas'."
            )

        return nome_tabela

        )
