@app.post("/pdv/finalizar")
def finalizar_pdv():

    dados = request.json

    empresa = dados.get("empresa")
    comanda = dados.get("comanda")
    usuario = dados.get("usuario", 1)
    itens = dados.get("itens", [])

    if not itens:
        return jsonify({"erro": "Nenhum item enviado"}), 400

    conn = None
    cur = None

    try:

        conn = get_connection()
        cur = conn.cursor()

        # Verificar se já existe orçamento aberto
        cur.execute("""
            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'D'
        """, (empresa, comanda))

        orc = cur.fetchone()

        if orc:
            novo_orc_id = orc[0]

        else:

            cur.execute("""
                SELECT SEQ_SEQUENCIAL
                FROM TB_SEQUENCIAL
                WHERE SEQ_EMPRESA = ?
                AND SEQ_TABELA = 'TB_ORCAMENTO'
                FOR UPDATE
            """, (empresa,))

            row = cur.fetchone()

            if not row:
                raise Exception("Sequencial TB_ORCAMENTO não encontrado")

            novo_orc_id = row[0] + 1

            cur.execute("""
                UPDATE TB_SEQUENCIAL
                SET SEQ_SEQUENCIAL = ?
                WHERE SEQ_EMPRESA = ?
                AND SEQ_TABELA = 'TB_ORCAMENTO'
            """, (novo_orc_id, empresa))

            cur.execute("""
                INSERT INTO TB_ORCAMENTO (
                    ORC_EMPRESA,
                    ORC_ID,
                    ORC_STATUS,
                    ORC_NR_ITENS,
                    ORC_SUB_TOTAL,
                    ORC_VALOR_TOTAL,
                    ORC_DT_LANCAMENTO,
                    ORC_DT_EMISSAO,
                    ORC_COMANDA,
                    ORC_TIPO_MOVIMENTO,
                    ORC_TIPO_VENDA,
                    ORC_CLIENTE,
                    ORC_USUARIO
                )
                VALUES (?, ?, 'D', 0, 0, 0, CURRENT_TIMESTAMP, CURRENT_DATE, ?, 'O', 'N', 1, ?)
            """, (
                empresa,
                novo_orc_id,
                comanda,
                usuario
            ))

        cur.execute("""
            SELECT MAX(ORCI_SEQUENCIAL)
            FROM TB_ORCAMENTO_ITEM
            WHERE ORCI_EMPRESA = ?
            AND ORCI_ORCAMENTO = ?
        """, (empresa, novo_orc_id))

        row = cur.fetchone()
        sequencial_item = row[0] if row[0] else 0

        for item in itens:

            produto_id = item["produto_id"]

            cur.execute("""
                SELECT PRD_UN_VENDA,
                       PRD_PRECO_VENDA
                FROM TB_PRODUTO
                WHERE PRD_EMPRESA = ?
                AND PRD_ID = ?
                AND PRD_STATUS = 'A'
            """, (empresa, produto_id))

            prod = cur.fetchone()

            if not prod:
                raise Exception(f"Produto {produto_id} inválido")

            unidade, valor_unitario = prod

            quantidade = Decimal(str(item["quantidade"]))
            valor_unitario = Decimal(str(valor_unitario))

            sequencial_item += 1

            sub_total = quantidade * valor_unitario

            cur.execute("""
                INSERT INTO TB_ORCAMENTO_ITEM (
                    ORCI_EMPRESA,
                    ORCI_ORCAMENTO,
                    ORCI_SEQUENCIAL,
                    ORCI_PRODUTO,
                    ORCI_UNIDADE,
                    ORCI_QUANTIDADE,
                    ORCI_VOLUMES,
                    ORCI_VALOR_UNITARIO,
                    ORCI_SUB_TOTAL,
                    ORCI_VALOR_TOTAL,
                    ORCI_STATUS,
                    ORCI_COMANDA
                )
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'A', ?)
            """, (
                empresa,
                novo_orc_id,
                sequencial_item,
                produto_id,
                unidade,
                quantidade,
                1,
                valor_unitario,
                sub_total,
                sub_total,
                comanda
            ))

        cur.execute("""
            SELECT 
                SUM(ORCI_QUANTIDADE),
                SUM(ORCI_VALOR_TOTAL)
            FROM TB_ORCAMENTO_ITEM
            WHERE ORCI_EMPRESA = ?
            AND ORCI_ORCAMENTO = ?
        """, (empresa, novo_orc_id))

        totais = cur.fetchone()

        total_itens = totais[0] or 0
        total_geral = totais[1] or 0

        cur.execute("""
            UPDATE TB_ORCAMENTO
            SET ORC_NR_ITENS = ?,
                ORC_SUB_TOTAL = ?,
                ORC_VALOR_TOTAL = ?
            WHERE ORC_EMPRESA = ?
            AND ORC_ID = ?
        """, (total_itens, total_geral, total_geral, empresa, novo_orc_id))

        conn.commit()

        return jsonify({
            "sucesso": True,
            "orcamento_id": novo_orc_id
        })

    except Exception as e:

        if conn:
            conn.rollback()

        return jsonify({"erro": str(e)}), 500

    finally:

        try:
            if cur:
                cur.close()
            if conn:
                conn.close()
        except:
            pass

            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'A'
        """, (empresa, comanda))

        orc = cur.fetchone()

        if orc:
            return jsonify({
                "erro": f"Comanda {comanda} está sendo utilizada no balcão."
            }), 409


        # ------------------------------------------------
        # 2️⃣ Procurar orçamento digitado (D)
        # ------------------------------------------------
        cur.execute("""
            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'D'
            ORDER BY ORC_ID DESC
        """, (empresa, comanda))

        orc = cur.fetchone()

        if not orc:
            return jsonify({
                "orc_id": None,
                "itens": []
            })

        orc_id = orc[0]


        # ------------------------------------------------
        # 3️⃣ Carregar itens do orçamento
        # ------------------------------------------------
        cur.execute("""
            SELECT
                I.ORCI_PRODUTO,
                P.PRD_DESCRICAO,
                P.PRD_PRECO_VENDA,
                I.ORCI_QUANTIDADE,
                I.ORCI_VALOR_TOTAL
            FROM TB_ORCAMENTO_ITEM I
            JOIN TB_PRODUTO P
              ON P.PRD_EMPRESA = I.ORCI_EMPRESA
             AND P.PRD_ID = I.ORCI_PRODUTO
            WHERE I.ORCI_EMPRESA = ?
            AND I.ORCI_ORCAMENTO = ?
            ORDER BY I.ORCI_SEQUENCIAL

                "valor_unitario": float(row[2]),

            })


        return jsonify({
            "orc_id": orc_id,
            "itens": itens
        })


    except Exception as e:

        return jsonify({"erro": str(e)}), 500


    finally:

        try:
            if cur:
                cur.close()

                conn.close()

@app.post("/pdv/finalizar")
def finalizar_pdv():

    dados = request.json

    empresa = dados.get("empresa")
    comanda = dados.get("comanda")
    usuario = dados.get("usuario", 1)
    itens = dados.get("itens", [])

    if not itens:
        return jsonify({"erro": "Nenhum item enviado"}), 400

    conn = None
    cur = None

    try:

        conn = get_connection()
        cur = conn.cursor()

        # Verificar se já existe orçamento aberto
        cur.execute("""
            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'D'
        """, (empresa, comanda))

        orc = cur.fetchone()

        if orc:
            novo_orc_id = orc[0]

        else:

            cur.execute("""
                SELECT SEQ_SEQUENCIAL
                FROM TB_SEQUENCIAL
                WHERE SEQ_EMPRESA = ?
                AND SEQ_TABELA = 'TB_ORCAMENTO'
                FOR UPDATE
            """, (empresa,))

            row = cur.fetchone()

            if not row:
                raise Exception("Sequencial TB_ORCAMENTO não encontrado")

            novo_orc_id = row[0] + 1

            cur.execute("""
                UPDATE TB_SEQUENCIAL
                SET SEQ_SEQUENCIAL = ?
                WHERE SEQ_EMPRESA = ?
                AND SEQ_TABELA = 'TB_ORCAMENTO'
            """, (novo_orc_id, empresa))

            cur.execute("""
                INSERT INTO TB_ORCAMENTO (
                    ORC_EMPRESA,
                    ORC_ID,
                    ORC_STATUS,
                    ORC_NR_ITENS,
                    ORC_SUB_TOTAL,
                    ORC_VALOR_TOTAL,
                    ORC_DT_LANCAMENTO,
                    ORC_DT_EMISSAO,
                    ORC_COMANDA,
                    ORC_TIPO_MOVIMENTO,
                    ORC_TIPO_VENDA,
                    ORC_CLIENTE,
                    ORC_USUARIO
                )
                VALUES (?, ?, 'D', 0, 0, 0, CURRENT_TIMESTAMP, CURRENT_DATE, ?, 'O', 'N', 1, ?)
            """, (
                empresa,
                novo_orc_id,
                comanda,
                usuario
            ))

        cur.execute("""
            SELECT MAX(ORCI_SEQUENCIAL)
            FROM TB_ORCAMENTO_ITEM
            WHERE ORCI_EMPRESA = ?
            AND ORCI_ORCAMENTO = ?
        """, (empresa, novo_orc_id))

        row = cur.fetchone()
        sequencial_item = row[0] if row[0] else 0

        for item in itens:

            produto_id = item["produto_id"]

            cur.execute("""
                SELECT PRD_UN_VENDA,
                       PRD_PRECO_VENDA
                FROM TB_PRODUTO
                WHERE PRD_EMPRESA = ?
                AND PRD_ID = ?
                AND PRD_STATUS = 'A'
            """, (empresa, produto_id))

            prod = cur.fetchone()

            if not prod:
                raise Exception(f"Produto {produto_id} inválido")

            unidade, valor_unitario = prod

            quantidade = Decimal(str(item["quantidade"]))
            valor_unitario = Decimal(str(valor_unitario))

            sequencial_item += 1

            sub_total = quantidade * valor_unitario

            cur.execute("""
                INSERT INTO TB_ORCAMENTO_ITEM (
                    ORCI_EMPRESA,
                    ORCI_ORCAMENTO,
                    ORCI_SEQUENCIAL,
                    ORCI_PRODUTO,
                    ORCI_UNIDADE,
                    ORCI_QUANTIDADE,
                    ORCI_VOLUMES,
                    ORCI_VALOR_UNITARIO,
                    ORCI_SUB_TOTAL,
                    ORCI_VALOR_TOTAL,
                    ORCI_STATUS,
                    ORCI_COMANDA
                )
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'A', ?)
            """, (
                empresa,
                novo_orc_id,
                sequencial_item,
                produto_id,
                unidade,
                quantidade,
                1,
                valor_unitario,
                sub_total,
                sub_total,
                comanda
            ))

        cur.execute("""
            SELECT 
                SUM(ORCI_QUANTIDADE),
                SUM(ORCI_VALOR_TOTAL)
            FROM TB_ORCAMENTO_ITEM
            WHERE ORCI_EMPRESA = ?
            AND ORCI_ORCAMENTO = ?
        """, (empresa, novo_orc_id))

        totais = cur.fetchone()

        total_itens = totais[0] or 0
        total_geral = totais[1] or 0

        cur.execute("""
            UPDATE TB_ORCAMENTO
            SET ORC_NR_ITENS = ?,
                ORC_SUB_TOTAL = ?,
                ORC_VALOR_TOTAL = ?
            WHERE ORC_EMPRESA = ?
            AND ORC_ID = ?
        """, (total_itens, total_geral, total_geral, empresa, novo_orc_id))

        conn.commit()

        return jsonify({
            "sucesso": True,
            "orcamento_id": novo_orc_id
        })

    except Exception as e:

        if conn:
            conn.rollback()

        return jsonify({"erro": str(e)}), 500

    finally:

        try:
            if cur:
                cur.close()
            if conn:
                conn.close()
        except:
            pass

@app.get("/pdv/comanda")
def buscar_comanda():

    cur = None

    try:

        empresa = int(request.args.get("empresa"))
        comanda = int(request.args.get("comanda"))

        conn = get_connection()
        cur = conn.cursor()

        # ------------------------------------------------
        # 1️⃣ Verificar se a comanda está em uso (A)
        # ------------------------------------------------
        cur.execute("""
            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'A'
        """, (empresa, comanda))

        orc = cur.fetchone()

        if orc:
            return jsonify({
                "erro": f"Comanda {comanda} está sendo utilizada no balcão."
            }), 409


        # ------------------------------------------------
        # 2️⃣ Procurar orçamento digitado (D)
        # ------------------------------------------------
        cur.execute("""
            SELECT FIRST 1 ORC_ID
            FROM TB_ORCAMENTO
            WHERE ORC_EMPRESA = ?
            AND ORC_COMANDA = ?
            AND ORC_STATUS = 'D'
            ORDER BY ORC_ID DESC
        """, (empresa, comanda))

        orc = cur.fetchone()

        if not orc:
            return jsonify({
                "orc_id": None,
                "itens": []
            })

        orc_id = orc[0]


        # ------------------------------------------------
        # 3️⃣ Carregar itens do orçamento
        # ------------------------------------------------
        cur.execute("""
            SELECT
                I.ORCI_PRODUTO,
                P.PRD_DESCRICAO,
                P.PRD_PRECO_VENDA,
                I.ORCI_QUANTIDADE,
                I.ORCI_VALOR_TOTAL
            FROM TB_ORCAMENTO_ITEM I
            JOIN TB_PRODUTO P
              ON P.PRD_EMPRESA = I.ORCI_EMPRESA
             AND P.PRD_ID = I.ORCI_PRODUTO
            WHERE I.ORCI_EMPRESA = ?
            AND I.ORCI_ORCAMENTO = ?
            ORDER BY I.ORCI_SEQUENCIAL
        """, (empresa, orc_id))

        itens = []

        for row in cur.fetchall():
            itens.append({
                "produto_id": row[0],
                "descricao": row[1],
                "valor_unitario": float(row[2]),
                "quantidade": float(row[3]),
                "valor_total": float(row[4])
            })


        return jsonify({
            "orc_id": orc_id,
            "itens": itens
        })


    except Exception as e:

        return jsonify({"erro": str(e)}), 500


    finally:

        try:
            if cur:
                cur.close()
            if conn:
                conn.close()

        """, (total_itens, total_geral, total_geral, empresa, novo_orc_id))
